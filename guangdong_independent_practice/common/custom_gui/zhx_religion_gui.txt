# The engine has no true "clear school" effect. When the invisible lifecycle

        religious_school = {

        }
    }
}

# The native scholar selector is defined at religion-group scope. Eastern
# religions outside Ritual Teaching therefore receive the underlying hard-coded
# button too. This inert top-layer button restores the panel, closes the banner
# symmetrically and owns the hitbox without mutating Buddhist or Shinto state.
custom_button = {
    name = zhx_non_lijiao_invite_school_blocker
    potential = {
        has_dlc = "Cradle of Civilization"
        religion_group = eastern
        NOT = { zhx_is_lijiao_country = yes }
    }
}

# Four readouts share one 28x24 anchor. The ranges are mutually exclusive, so
# exactly one tier-coloured value is drawn and no text can stack over the school
# row. A later transparent custom button owns both hover and click input.
custom_text_box = {
    name = zhx_religion_practice_hollow_value
    potential = {
        zhx_is_lijiao_country = yes
        zhx_has_doctrine = yes
        NOT = {
            check_variable = { which = zhx_doctrine_practice value = 25 }
        }
    }
    tooltip = zhx_religion_practice_value_tt
}

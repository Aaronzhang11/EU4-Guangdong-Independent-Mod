# The Ritual Teaching tripod replaces the native Islamic scholar control and is
# the independent entry for the complete doctrine-practice rulebook. The number
# itself remains hover-only and never needs a transparent click layer.
custom_button = {
    name = zhx_lijiao_school_rules_button
    potential = {
        zhx_is_lijiao_country = yes
        zhx_has_doctrine = yes
        has_religious_school = yes
    }
    trigger = { always = yes }
    effect = { country_event = { id = zhx_doctrine.20 } }
    tooltip = zhx_religion_practice_rules_button_tt
}

# The engine has no true "clear school" effect. When the invisible lifecycle
# sentinel is active, this neutral plate hides the otherwise hard-coded Islamic
# invite-scholar artwork without presenting a false doctrine emblem.
custom_icon = {
    name = zhx_no_doctrine_school_button_overlay
    potential = {
        religious_school = {
            group = eastern
            school = zhx_no_doctrine_school
        }
    }
}

# Four readouts share one 28x24 anchor. The ranges are mutually exclusive, so
# exactly one tier-coloured value is drawn and no text can stack over the school
# row. The text box itself owns the concise active-factor tooltip.
custom_text_box = {
    name = zhx_religion_practice_hollow_value
    potential = {
        zhx_is_lijiao_country = yes
        zhx_has_doctrine = yes
        NOT = {
            check_variable = { which = zhx_doctrine_practice value = 25 }

    }

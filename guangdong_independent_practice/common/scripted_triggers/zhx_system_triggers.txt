# 周天下系统的公共判断接口。玩法、事件和 GUI 应调用这些接口，
# 不要在各处直接写死 CZH 或底层国家旗标。
zhx_is_tianzi = {
    # 当前作用域国家是否为天子。
    has_country_flag = zhx_tianzi
}

zhx_is_tianxia_member = {
    # 当前作用域国家是否为普通周天下成员。
    has_country_flag = zhx_member
}

zhx_is_major_feudatory = {
    # 大诸侯同时也是普通成员；此旗标只表示额外的政治身份。
    has_country_flag = zhx_member
    has_country_flag = zhx_major_feudatory
}

# 权威条所称“七大诸侯”读取可动态授予和撤销的大诸侯身份。
# 不绑定任何国家标签，后续改选七国时只需变更 zhx_major_feudatory 旗标。
zhx_is_seven_great_feudatory = {
    zhx_is_major_feudatory = yes
}

zhx_is_tianxia_polity = {
    # “周天下国家”包括天子本人和所有诸侯成员。
    OR = {
        has_country_flag = zhx_tianzi
        has_country_flag = zhx_member
    }
}

# The three Mandate-city modifiers are evaluated before on_startup events.
# During that brief pre-initialisation window the authoritative membership
# flags do not exist yet, so recognise the three known 1444 member controllers.
# Once v7 is initialised, the normal dynamic Zhou flags are the only authority.
zhx_is_tianxia_city_controller = {
    OR = {
        zhx_is_tianxia_polity = yes
        AND = {
            NOT = { has_global_flag = zhx_system_initialised_v7 }
            OR = {
                tag = YAN
                tag = WUU
                tag = GDD
            }
        }
    }
}

# 第一版的“天下无战事”：任何仍然存在的周天下国家都没有参加战争。
# 目前连对外战争也会打破和平；后续战争层再细分诸侯内战、对外进攻与对外防御。
zhx_tianxia_is_at_peace = {
    NOT = {
        any_country = {
            exists = yes
            zhx_is_tianxia_polity = yes
            is_at_war = yes
        }
    }
}

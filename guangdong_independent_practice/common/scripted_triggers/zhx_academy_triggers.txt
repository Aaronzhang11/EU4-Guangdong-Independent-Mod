# School-level ownership checks. Two academies of the same school collapse to
# one boolean so neither national synergy nor thought tension scales by count.

zhx_has_ru_academy = {
    any_owned_province = {
        OR = {
            has_province_modifier = zhx_academy_chongli
            has_province_modifier = zhx_academy_minglun
        }
    }
}

# A school is protected by either the state's main doctrine or its currently
# invited secondary school, or a temporary old-school residue created by a
# main-doctrine reform.  The latter two prevent tension but never grant the
# matching-academy national synergy.

zhx_has_unprotected_ru_academy = {
    zhx_has_ru_academy = yes
    NOT = { has_country_flag = zhx_doctrine_ru }
    NOT = { has_country_modifier = zhx_ru_invited_scholar_modifier }
    NOT = { zhx_doctrine_reform_protects_ru_academies = yes }
}

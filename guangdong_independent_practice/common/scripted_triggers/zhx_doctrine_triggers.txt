# Eligibility and authoritative doctrine-state checks for the first
# Ritual Teaching / Hundred Schools vertical slice.

zhx_is_lijiao_country = {
    religion = confucianism
}

zhx_has_doctrine = {
    OR = {
        has_country_flag = zhx_doctrine_ru
        has_country_flag = zhx_doctrine_fa
        has_country_flag = zhx_doctrine_mo
    }
}

# All six reserved doctrine flags. Only Ru, Fa and Mo are playable in the
# current vertical slice; the broader trigger keeps conversion cleanup complete
# as the other three schools come online.
zhx_has_any_doctrine_flag = {
    OR = {
        has_country_flag = zhx_doctrine_ru
        has_country_flag = zhx_doctrine_fa
        has_country_flag = zhx_doctrine_mo
        has_country_flag = zhx_doctrine_dao
        has_country_flag = zhx_doctrine_bing
        has_country_flag = zhx_doctrine_zongheng
    }
}

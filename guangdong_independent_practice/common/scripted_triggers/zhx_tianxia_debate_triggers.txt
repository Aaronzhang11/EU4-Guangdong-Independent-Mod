# Cheap UI gate: never iterate over all countries from a custom-GUI refresh.
# The authoritative social-basis scan remains in the click/AI effect below.
zhx_can_attempt_tianxia_debate_petition = {
    exists = yes
    zhx_is_tianxia_polity = yes
    zhx_is_lijiao_country = yes
    zhx_doctrine_reform_may_file_tianxia_proposal = yes
    check_variable = { which = zhx_doctrine_practice value = 70 }
    has_saved_global_event_target = zhx_tianzi
    event_target:zhx_tianzi = {
        NOT = { has_country_modifier = zhx_tianxia_orthodoxy_term }
        NOT = { has_country_flag = zhx_council_kind_debate }
    }
    OR = {
        AND = { has_country_flag = zhx_doctrine_ru event_target:zhx_tianzi = { NOT = { has_country_modifier = zhx_debate_petition_ru } } }
        AND = { has_country_flag = zhx_doctrine_fa event_target:zhx_tianzi = { NOT = { has_country_modifier = zhx_debate_petition_fa } } }
        AND = { has_country_flag = zhx_doctrine_mo event_target:zhx_tianzi = { NOT = { has_country_modifier = zhx_debate_petition_mo } } }
        AND = { has_country_flag = zhx_doctrine_dao event_target:zhx_tianzi = { NOT = { has_country_modifier = zhx_debate_petition_dao } } }
        AND = { has_country_flag = zhx_doctrine_bing event_target:zhx_tianzi = { NOT = { has_country_modifier = zhx_debate_petition_bing } } }
        AND = { has_country_flag = zhx_doctrine_zongheng event_target:zhx_tianzi = { NOT = { has_country_modifier = zhx_debate_petition_zongheng } } }
    }
}

# Reaching 70 grants only the right to attempt filing. The three-adopter and
# independent 50-practice endorser checks run once on click or in the yearly AI
# attempt, never once per rendered GUI frame.
zhx_can_file_tianxia_debate_petition = {
    zhx_can_attempt_tianxia_debate_petition = yes
    OR = {
        AND = { has_country_flag = zhx_doctrine_ru zhx_tianxia_ru_has_social_basis = yes }
        AND = { has_country_flag = zhx_doctrine_fa zhx_tianxia_fa_has_social_basis = yes }
        AND = { has_country_flag = zhx_doctrine_mo zhx_tianxia_mo_has_social_basis = yes }
        AND = { has_country_flag = zhx_doctrine_dao zhx_tianxia_dao_has_social_basis = yes }
        AND = { has_country_flag = zhx_doctrine_bing zhx_tianxia_bing_has_social_basis = yes }
        AND = { has_country_flag = zhx_doctrine_zongheng zhx_tianxia_zongheng_has_social_basis = yes }
    }
}

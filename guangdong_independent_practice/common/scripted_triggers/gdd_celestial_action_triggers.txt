# Shared authorization used by every replacement decree and reform row.
# Authority decides which country may act, but never restricts that country
# to either the centralising or decentralising reform branch.
gdd_celestial_action_authorized_trigger = {
    custom_trigger_tooltip = {
        tooltip = GDD_CELESTIAL_ACTION_AUTHORIZED_TRIGGER_TT
        OR = {
            AND = {
                has_saved_global_event_target = gdd_principal_vassal
                tag = event_target:gdd_principal_vassal
                CZH = {
                    NOT = {
                        check_variable = {
                            which = gdd_authority_integer_display
                            value = -25
                        }
                    }
                }
            }
            AND = {
                is_emperor_of_china = yes
                CZH = {
                    check_variable = {
                        which = gdd_authority_integer_display
                        value = 26
                    }
                }
            }
        }
    }
}

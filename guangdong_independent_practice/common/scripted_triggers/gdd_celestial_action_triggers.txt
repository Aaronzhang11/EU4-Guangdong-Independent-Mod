# Shared authorization used by every replacement decree and reform row.
# YAN remains the temporary principal-vassal test country; the Emperor keeps
# access until the future authority-balance mechanic decides between them.
gdd_celestial_action_authorized_trigger = {
    OR = {
        tag = YAN
        is_emperor_of_china = yes
    }
}

gdd_can_enact_proxy_establish_gaituguiliu_trigger = {
    NOT = { gdd_reform_establish_gaituguiliu_passed = yes }
    if = {
        limit = { has_estate = estate_eunuchs }
        OR = {
            estate_loyalty = {
                estate = estate_eunuchs
                loyalty = 60
            }
            NOT = { corruption = 1 }
        }
    }
    else = { NOT = { corruption = 1 } }
}

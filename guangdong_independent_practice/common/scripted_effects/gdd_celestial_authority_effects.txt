# Gameplay layer for the independent central-authority / feudatory-prestige
# balance shown in the Mandate window.  The authoritative value and all cached
# display values live on CZH.  Public scripts that change the balance must call
# gdd_refresh_authority_balance_effects after changing it.

gdd_clamp_central_authority_balance = {
    if = {
        limit = {
            check_variable = {
                which = gdd_central_authority_balance
                value = 100
            }
        }
        set_variable = {
            which = gdd_central_authority_balance
            value = 100
        }
    }
    if = {
        limit = {
            NOT = {
                check_variable = {
                    which = gdd_central_authority_balance
                    value = -100
                }
            }
        }
        set_variable = {
            which = gdd_central_authority_balance
            value = -100
        }
    }

    # Tenth-point changes can leave a microscopic floating-point residue after
    # opposite adjustments. Treat the supported one-decimal zero band as zero.
    if = {
        limit = {
            check_variable = {
                which = gdd_central_authority_balance
                value = -0.05
            }
            NOT = {
                check_variable = {
                    which = gdd_central_authority_balance
                    value = 0.05
                }
            }
        }
        set_variable = {
            which = gdd_central_authority_balance
            value = 0
        }
    }
}

# CZH-scope rebuild. Seven whole-number bits plus four tenth-point bits
# reproduce every value from 0.0 through 100.0 at one-decimal precision.
gdd_refresh_authority_balance_effects = {
    gdd_clamp_central_authority_balance = yes

    clr_country_flag = gdd_central_authority_active
    clr_country_flag = gdd_feudatory_prestige_active
    clr_country_flag = gdd_authority_balance_zero_active
    clr_country_flag = gdd_authority_balance_bit_001
    clr_country_flag = gdd_authority_balance_bit_002
    clr_country_flag = gdd_authority_balance_bit_004
    clr_country_flag = gdd_authority_balance_bit_008
    clr_country_flag = gdd_authority_balance_bit_016
    clr_country_flag = gdd_authority_balance_bit_032
    clr_country_flag = gdd_authority_balance_bit_064
    clr_country_flag = gdd_authority_balance_tenth_bit_001
    clr_country_flag = gdd_authority_balance_tenth_bit_002
    clr_country_flag = gdd_authority_balance_tenth_bit_004
    clr_country_flag = gdd_authority_balance_tenth_bit_008

    set_variable = { which = gdd_central_authority_display value = 0 }
    set_variable = { which = gdd_feudatory_prestige_display value = 0 }
    set_variable = { which = gdd_authority_half_display value = 0 }
    set_variable = { which = gdd_authority_tenth_display value = 0 }
    set_variable = { which = gdd_authority_quarter_display value = 0 }
    set_variable = { which = gdd_authority_effect_remainder value = 0 }
    set_variable = { which = gdd_authority_integer_magnitude_display value = 0 }
    set_variable = { which = gdd_authority_integer_display value = 0 }

    if = {
        limit = {
            is_variable_equal = {
                which = gdd_central_authority_balance
                value = 0
            }
        }
        set_country_flag = gdd_authority_balance_zero_active
    }
    else_if = {
        limit = {
            check_variable = {
                which = gdd_central_authority_balance
                value = 0
            }
        }
        set_country_flag = gdd_central_authority_active
        set_variable = {
            which = gdd_central_authority_display
            which = gdd_central_authority_balance
        }
        set_variable = {
            which = gdd_authority_effect_remainder
            which = gdd_central_authority_balance
        }
    }
    else = {
        set_country_flag = gdd_feudatory_prestige_active
        set_variable = {
            which = gdd_feudatory_prestige_display
            which = gdd_central_authority_balance
        }
        multiply_variable = {
            which = gdd_feudatory_prestige_display
            value = -1
        }
        set_variable = {
            which = gdd_authority_effect_remainder
            which = gdd_feudatory_prestige_display
        }
    }

    if = {
        limit = { has_country_flag = gdd_central_authority_active }
        set_variable = {
            which = gdd_authority_half_display
            which = gdd_central_authority_display
        }

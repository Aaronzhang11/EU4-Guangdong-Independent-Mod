# Adapter between the independent reform state machine and the mature doctrine,
# native-school, academy and Tianxia layers.  Reform effects set output flags;
# this consumer performs each expensive refresh once, then clears every port.
zhx_consume_doctrine_reform_integration_hooks = {
    if = {
        limit = { has_country_flag = zhx_doctrine_reform_hook_clear_invited_school }
        zhx_clear_invited_school_modifiers = yes
    }
    if = {
        limit = {
            OR = {
                has_country_flag = zhx_doctrine_reform_hook_main_school_changed
                has_country_flag = zhx_doctrine_reform_hook_refresh_doctrine_presentation
            }
        }
        zhx_sync_native_doctrine_school = yes
        zhx_clamp_doctrine_practice = yes
        zhx_refresh_doctrine_tier = yes
        zhx_prepare_doctrine_ledger = yes
    }
    if = {
        limit = {
            has_country_flag = zhx_doctrine_reform_hook_cancel_pending_tianxia_petition
            has_saved_global_event_target = zhx_tianzi
        }
        event_target:zhx_tianzi = { zhx_cleanup_tianxia_debate_petitions = yes }
    }
    if = {
        limit = { has_country_flag = zhx_doctrine_reform_hook_refresh_academies }
        zhx_refresh_academy_country_effects = yes
    }
    zhx_doctrine_reform_acknowledge_integration_hooks = yes
}

# One authoritative calculation serves both the real yearly settlement and the
# read-only hover ledger. It only writes a scratch value; callers decide whether
# to apply it to practice or merely cache it for presentation.
zhx_calculate_doctrine_yearly_delta = {
    set_variable = {
        which = zhx_doctrine_calculated_delta
        value = 0
    }

    # Ru: orderly government, public standing and restraint of exhaustion.
    if = {
        limit = { has_country_flag = zhx_doctrine_ru }
        if = {
            limit = { stability = 1 }
            change_variable = { which = zhx_doctrine_calculated_delta value = 2 }
            if = {
                limit = { has_country_flag = zhx_doctrine_hover_cache_building }
                set_country_flag = zhx_doctrine_hover_positive_1
            }
        }
        else_if = {
            limit = { NOT = { stability = 0 } }
            change_variable = { which = zhx_doctrine_calculated_delta value = -3 }
            if = {
                limit = { has_country_flag = zhx_doctrine_hover_cache_building }
                set_country_flag = zhx_doctrine_hover_negative_1
            }
        }
        if = {
            limit = { prestige = 25 }
            change_variable = { which = zhx_doctrine_calculated_delta value = 1 }
            if = {
                limit = { has_country_flag = zhx_doctrine_hover_cache_building }
                set_country_flag = zhx_doctrine_hover_positive_2
            }
        }
        else_if = {
            limit = { NOT = { prestige = -25 } }
            change_variable = { which = zhx_doctrine_calculated_delta value = -1 }
            if = {
                limit = { has_country_flag = zhx_doctrine_hover_cache_building }
                set_country_flag = zhx_doctrine_hover_negative_2
            }
        }
        if = {
            limit = { NOT = { war_exhaustion = 3 } }
            change_variable = { which = zhx_doctrine_calculated_delta value = 1 }
            if = {
                limit = { has_country_flag = zhx_doctrine_hover_cache_building }
                set_country_flag = zhx_doctrine_hover_positive_3
            }
        }
        else_if = {
            limit = { war_exhaustion = 5 }
            change_variable = { which = zhx_doctrine_calculated_delta value = -2 }
            if = {
                limit = { has_country_flag = zhx_doctrine_hover_cache_building }
                set_country_flag = zhx_doctrine_hover_negative_3
            }
        }
    }

    # Fa: clear laws are measured through corruption, autonomy and overextension.
    else_if = {
        limit = { has_country_flag = zhx_doctrine_fa }
        if = {
            limit = { NOT = { corruption = 2 } }
            change_variable = { which = zhx_doctrine_calculated_delta value = 2 }
            if = {
                limit = { has_country_flag = zhx_doctrine_hover_cache_building }
                set_country_flag = zhx_doctrine_hover_positive_1
            }
        }
        else_if = {
            limit = { corruption = 5 }
            change_variable = { which = zhx_doctrine_calculated_delta value = -3 }
            if = {
                limit = { has_country_flag = zhx_doctrine_hover_cache_building }
                set_country_flag = zhx_doctrine_hover_negative_1
            }
        }
        if = {
            limit = { NOT = { average_autonomy_above_min = 25 } }
            change_variable = { which = zhx_doctrine_calculated_delta value = 1 }
            if = {
                limit = { has_country_flag = zhx_doctrine_hover_cache_building }
                set_country_flag = zhx_doctrine_hover_positive_2
            }
        }
        else_if = {
            limit = { average_autonomy_above_min = 50 }
            change_variable = { which = zhx_doctrine_calculated_delta value = -2 }
            if = {
                limit = { has_country_flag = zhx_doctrine_hover_cache_building }
                set_country_flag = zhx_doctrine_hover_negative_2
            }
        }
        if = {
            limit = { NOT = { overextension_percentage = 0.10 } }
            change_variable = { which = zhx_doctrine_calculated_delta value = 1 }
            if = {
                limit = { has_country_flag = zhx_doctrine_hover_cache_building }
                set_country_flag = zhx_doctrine_hover_positive_3
            }
        }
        else_if = {
            limit = { overextension_percentage = 0.25 }
            change_variable = { which = zhx_doctrine_calculated_delta value = -2 }
            if = {
                limit = { has_country_flag = zhx_doctrine_hover_cache_building }
                set_country_flag = zhx_doctrine_hover_negative_3
            }
        }
    }

    # Mo: non-aggression, defensive war and frugality are concrete conduct.
    else_if = {
        limit = { has_country_flag = zhx_doctrine_mo }
        if = {
            limit = {
                is_in_war = { defender_leader = ROOT }
            }
            change_variable = { which = zhx_doctrine_calculated_delta value = 3 }
            if = {
                limit = { has_country_flag = zhx_doctrine_hover_cache_building }
                set_country_flag = zhx_doctrine_hover_mo_defender
            }
        }
        else_if = {
            limit = { is_at_war = no }
            change_variable = { which = zhx_doctrine_calculated_delta value = 2 }
            if = {
                limit = { has_country_flag = zhx_doctrine_hover_cache_building }
                set_country_flag = zhx_doctrine_hover_mo_peace
            }
        }
        if = {
            limit = { NOT = { num_of_loans = 1 } }
            change_variable = { which = zhx_doctrine_calculated_delta value = 1 }
            if = {
                limit = { has_country_flag = zhx_doctrine_hover_cache_building }
                set_country_flag = zhx_doctrine_hover_positive_2
            }
        }
        else_if = {
            limit = { num_of_loans = 3 }
            change_variable = { which = zhx_doctrine_calculated_delta value = -2 }
            if = {
                limit = { has_country_flag = zhx_doctrine_hover_cache_building }
                set_country_flag = zhx_doctrine_hover_negative_2
            }
        }
        if = {
            limit = { NOT = { war_exhaustion = 3 } }
            change_variable = { which = zhx_doctrine_calculated_delta value = 1 }
            if = {
                limit = { has_country_flag = zhx_doctrine_hover_cache_building }
                set_country_flag = zhx_doctrine_hover_positive_3
            }
        }
        else_if = {
            limit = { war_exhaustion = 5 }
            change_variable = { which = zhx_doctrine_calculated_delta value = -2 }
            if = {
                limit = { has_country_flag = zhx_doctrine_hover_cache_building }
                set_country_flag = zhx_doctrine_hover_negative_3
            }
        }
        if = {
            limit = {
                is_in_war = { attacker_leader = ROOT }
            }
            change_variable = { which = zhx_doctrine_calculated_delta value = -4 }
            if = {
                limit = { has_country_flag = zhx_doctrine_hover_cache_building }
                set_country_flag = zhx_doctrine_hover_negative_1
            }
        }
    }

# Refresh the country-local hover ledger cache. The monthly pulse calls this
# only for countries with a live doctrine; adoption, opening setup and annual
# settlement also call it immediately so the religion view is never born stale.
zhx_prepare_doctrine_ledger = {
    zhx_clear_doctrine_hover_cache = yes
    set_country_flag = zhx_doctrine_hover_cache_building
    zhx_calculate_doctrine_yearly_delta = yes
    clr_country_flag = zhx_doctrine_hover_cache_building

    # The hover explains how the currently-triggered rows add up. Keep the raw
    # formula total here instead of applying the 0-100 practice clamp; the UI
    # intentionally omits a synthetic cap row, so its visible arithmetic must
    # equal the sum of the listed factors.
    set_variable = {
        which = zhx_doctrine_ledger_estimated_delta
        which = zhx_doctrine_calculated_delta
    }
    if = {
        limit = {
            NOT = {
                check_variable = { which = zhx_doctrine_practice value = 25 }
            }
        }
        set_variable = { which = zhx_doctrine_ledger_to_next_tier value = 25 }
        subtract_variable = {
            which = zhx_doctrine_ledger_to_next_tier
            which = zhx_doctrine_practice
        }
    }
    else_if = {
        limit = {
            NOT = {
                check_variable = { which = zhx_doctrine_practice value = 50 }
            }
        }
        set_variable = { which = zhx_doctrine_ledger_to_next_tier value = 50 }
        subtract_variable = {
            which = zhx_doctrine_ledger_to_next_tier
            which = zhx_doctrine_practice
        }
    }
    else_if = {
        limit = {
            NOT = {
                check_variable = { which = zhx_doctrine_practice value = 75 }
            }
        }
        set_variable = { which = zhx_doctrine_ledger_to_next_tier value = 75 }
        subtract_variable = {
            which = zhx_doctrine_ledger_to_next_tier
            which = zhx_doctrine_practice
        }
    }
    else = {
        set_variable = { which = zhx_doctrine_ledger_to_next_tier value = 0 }
    }

    # A doctrine needs 70 practice before its state can support a Tianxia
    # proposal. This is a read-only cached gap; polity eligibility remains a
    # separate requirement owned by the Tianxia system.
    set_variable = { which = zhx_doctrine_ledger_to_proposal value = 70 }
    if = {
        limit = {
            check_variable = { which = zhx_doctrine_practice value = 70 }
        }
        set_variable = { which = zhx_doctrine_ledger_to_proposal value = 0 }
    }
    else = {
        subtract_variable = {
            which = zhx_doctrine_ledger_to_proposal
            which = zhx_doctrine_practice
        }
    }
}

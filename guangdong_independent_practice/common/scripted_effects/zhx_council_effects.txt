zhx_reset_tianxia_council_ballot = {
    clr_country_flag = zhx_council_batching_ai_votes
    set_variable = { which = zhx_council_vote_a_count value = 0 }
    set_variable = { which = zhx_council_vote_b_count value = 0 }
    set_variable = { which = zhx_council_vote_c_count value = 0 }
    set_variable = { which = zhx_council_a_vs_b value = 0 }
    set_variable = { which = zhx_council_a_vs_c value = 0 }
    set_variable = { which = zhx_council_b_vs_a value = 0 }
    set_variable = { which = zhx_council_b_vs_c value = 0 }
    set_variable = { which = zhx_council_eligible_country_count value = 0 }
    set_variable = { which = zhx_council_voted_country_count value = 0 }
    set_variable = { which = zhx_council_unvoted_country_count value = 0 }
    set_variable = { which = zhx_council_eligible_ai_count value = 0 }
    set_variable = { which = zhx_council_voted_ai_count value = 0 }
    set_variable = { which = zhx_council_unvoted_ai_count value = 0 }
    every_country = {
        limit = {
            OR = {
                has_country_flag = zhx_council_vote_a
                has_country_flag = zhx_council_vote_b
                has_country_flag = zhx_council_vote_c
                has_country_flag = zhx_council_vote_bonus
                has_country_flag = zhx_council_ai_first_vote_pending
                has_country_flag = zhx_council_ai_review_pending
            }
        }
        zhx_clear_current_tianxia_council_vote = yes
        zhx_clear_tianxia_council_ai_schedule = yes
    }
}

zhx_recount_tianxia_council_ballot = {
    # A cast effect enters the Tianzi scope through event_target, but ROOT still
    # points at the casting country. Every public counter therefore names the
    # Tianzi explicitly; otherwise AI ballots are silently counted on the AI.
    event_target:zhx_tianzi = {
        set_variable = { which = zhx_council_vote_a_count value = 0 }
        set_variable = { which = zhx_council_vote_b_count value = 0 }
        set_variable = { which = zhx_council_vote_c_count value = 0 }
        set_variable = { which = zhx_council_eligible_country_count value = 0 }
        set_variable = { which = zhx_council_voted_country_count value = 0 }
        set_variable = { which = zhx_council_unvoted_country_count value = 0 }
        set_variable = { which = zhx_council_eligible_ai_count value = 0 }
        set_variable = { which = zhx_council_voted_ai_count value = 0 }
        set_variable = { which = zhx_council_unvoted_ai_count value = 0 }
    }

    every_country = {
        limit = {
            exists = yes
            OR = {
                AND = { event_target:zhx_tianzi = { has_country_flag = zhx_council_kind_ritual_breakdown } zhx_is_tianxia_member = yes }
                AND = { event_target:zhx_tianzi = { has_country_flag = zhx_council_kind_debate } zhx_is_tianxia_polity = yes }
            }
        }
        event_target:zhx_tianzi = { change_variable = { which = zhx_council_eligible_country_count value = 1 } }
        if = {
            limit = { ai = yes }
            event_target:zhx_tianzi = { change_variable = { which = zhx_council_eligible_ai_count value = 1 } }
        }
        if = {
            limit = {
                OR = {
                    has_country_flag = zhx_council_vote_a
                    has_country_flag = zhx_council_vote_b
                    has_country_flag = zhx_council_vote_c
                }
            }
            event_target:zhx_tianzi = { change_variable = { which = zhx_council_voted_country_count value = 1 } }
            if = {
                limit = { ai = yes }
                event_target:zhx_tianzi = { change_variable = { which = zhx_council_voted_ai_count value = 1 } }
            }
        }
    }

    every_country = {
        limit = {
            exists = yes
            has_country_flag = zhx_council_vote_a
            OR = {
                AND = { event_target:zhx_tianzi = { has_country_flag = zhx_council_kind_ritual_breakdown } zhx_is_tianxia_member = yes }
                AND = { event_target:zhx_tianzi = { has_country_flag = zhx_council_kind_debate } zhx_is_tianxia_polity = yes }
            }
        }
        event_target:zhx_tianzi = { change_variable = { which = zhx_council_vote_a_count value = 1 } }
        if = {
            limit = { has_country_flag = zhx_council_vote_bonus }
            event_target:zhx_tianzi = { change_variable = { which = zhx_council_vote_a_count value = 1 } }
        }
    }
    every_country = {
        limit = {
            exists = yes
            has_country_flag = zhx_council_vote_b
            OR = {
                AND = { event_target:zhx_tianzi = { has_country_flag = zhx_council_kind_ritual_breakdown } zhx_is_tianxia_member = yes }
                AND = { event_target:zhx_tianzi = { has_country_flag = zhx_council_kind_debate } zhx_is_tianxia_polity = yes }
            }
        }
        event_target:zhx_tianzi = { change_variable = { which = zhx_council_vote_b_count value = 1 } }
        if = {
            limit = { has_country_flag = zhx_council_vote_bonus }
            event_target:zhx_tianzi = { change_variable = { which = zhx_council_vote_b_count value = 1 } }
        }
    }
    every_country = {
        limit = {
            exists = yes
            has_country_flag = zhx_council_vote_c
            OR = {
                AND = { event_target:zhx_tianzi = { has_country_flag = zhx_council_kind_ritual_breakdown } zhx_is_tianxia_member = yes }
                AND = { event_target:zhx_tianzi = { has_country_flag = zhx_council_kind_debate } zhx_is_tianxia_polity = yes }
            }
        }
        event_target:zhx_tianzi = { change_variable = { which = zhx_council_vote_c_count value = 1 } }
    }

    event_target:zhx_tianzi = {
        set_variable = { which = zhx_council_unvoted_country_count which = zhx_council_eligible_country_count }
        subtract_variable = { which = zhx_council_unvoted_country_count which = zhx_council_voted_country_count }
        set_variable = { which = zhx_council_unvoted_ai_count which = zhx_council_eligible_ai_count }
        subtract_variable = { which = zhx_council_unvoted_ai_count which = zhx_council_voted_ai_count }
        set_variable = { which = zhx_council_a_vs_b which = zhx_council_vote_a_count }
        subtract_variable = { which = zhx_council_a_vs_b which = zhx_council_vote_b_count }
        set_variable = { which = zhx_council_a_vs_c which = zhx_council_vote_a_count }
        subtract_variable = { which = zhx_council_a_vs_c which = zhx_council_vote_c_count }
        set_variable = { which = zhx_council_b_vs_a which = zhx_council_vote_b_count }
        subtract_variable = { which = zhx_council_b_vs_a which = zhx_council_vote_a_count }
        set_variable = { which = zhx_council_b_vs_c which = zhx_council_vote_b_count }
        subtract_variable = { which = zhx_council_b_vs_c which = zhx_council_vote_c_count }
    }
}

zhx_cast_tianxia_council_vote_a = {
    if = {
        limit = { zhx_can_vote_in_tianxia_council = yes }
        zhx_clear_current_tianxia_council_vote = yes
        set_country_flag = zhx_council_vote_a
        set_variable = { which = zhx_council_vote_weight value = 1 }
        if = {
            limit = {
                check_variable = { which = zhx_doctrine_practice value = 75 }
                event_target:zhx_tianzi = { has_country_flag = zhx_council_kind_debate }
                zhx_council_vote_matches_candidate_a = yes
            }
            set_country_flag = zhx_council_vote_bonus
            set_variable = { which = zhx_council_vote_weight value = 2 }
        }
        if = {
            limit = {
                event_target:zhx_tianzi = {
                    NOT = { has_country_flag = zhx_council_batching_ai_votes }
                }
            }
            event_target:zhx_tianzi = { zhx_recount_tianxia_council_ballot = yes }
        }
    }
}

# Runs in one AI polity's country scope. The first decisive preference wins:
# shared school, petition authorship, one-sided friendship, one-sided hostility,
# then pluralism for an excluded school. Only truly neutral countries randomise.
zhx_choose_tianxia_council_ai_vote = {
    if = {
        limit = {
            ai = yes
            zhx_can_vote_in_tianxia_council = yes
            event_target:zhx_tianzi = {
                has_country_flag = zhx_council_kind_debate
                zhx_council_has_valid_candidates = yes
            }
        }
        if = {
            limit = { zhx_council_vote_matches_candidate_a = yes }
            zhx_cast_tianxia_council_vote_a = yes
        }
        else_if = {
            limit = { zhx_council_vote_matches_candidate_b = yes }
            zhx_cast_tianxia_council_vote_b = yes
        }
        else_if = {
            limit = { zhx_council_is_candidate_a_proposer = yes }
            zhx_cast_tianxia_council_vote_a = yes
        }
        else_if = {
            limit = { zhx_council_is_candidate_b_proposer = yes }
            zhx_cast_tianxia_council_vote_b = yes
        }
        else_if = {
            limit = {
                OR = {
                    zhx_council_is_allied_to_candidate_a_proposer = yes
                    zhx_council_likes_candidate_a_proposer = yes
                }
                NOT = {
                    OR = {
                        zhx_council_is_allied_to_candidate_b_proposer = yes
                        zhx_council_likes_candidate_b_proposer = yes
                        zhx_council_rivals_candidate_a_proposer = yes
                        zhx_council_is_at_war_with_candidate_a_proposer = yes
                    }
                }
            }
            zhx_cast_tianxia_council_vote_a = yes
        }
        else_if = {
            limit = {
                OR = {
                    zhx_council_is_allied_to_candidate_b_proposer = yes
                    zhx_council_likes_candidate_b_proposer = yes
                }
                NOT = {
                    OR = {
                        zhx_council_is_allied_to_candidate_a_proposer = yes
                        zhx_council_likes_candidate_a_proposer = yes
                        zhx_council_rivals_candidate_b_proposer = yes
                        zhx_council_is_at_war_with_candidate_b_proposer = yes
                    }
                }
            }
            zhx_cast_tianxia_council_vote_b = yes
        }
        else_if = {
            limit = {
                OR = {
                    zhx_council_rivals_candidate_a_proposer = yes
                    zhx_council_is_at_war_with_candidate_a_proposer = yes
                }
                NOT = {
                    OR = {
                        zhx_council_rivals_candidate_b_proposer = yes
                        zhx_council_is_at_war_with_candidate_b_proposer = yes
                    }
                }
            }
            zhx_cast_tianxia_council_vote_b = yes
        }
        else_if = {
            limit = {
                OR = {
                    zhx_council_rivals_candidate_b_proposer = yes
                    zhx_council_is_at_war_with_candidate_b_proposer = yes
                }
                NOT = {
                    OR = {
                        zhx_council_rivals_candidate_a_proposer = yes
                        zhx_council_is_at_war_with_candidate_a_proposer = yes
                    }
                }
            }
            zhx_cast_tianxia_council_vote_a = yes
        }
        else_if = {
            limit = { zhx_council_own_school_is_not_nominated = yes }
            zhx_cast_tianxia_council_vote_c = yes
        }
        else = {
            random_list = {
                40 = { zhx_cast_tianxia_council_vote_a = yes }
                40 = { zhx_cast_tianxia_council_vote_b = yes }
                20 = { zhx_cast_tianxia_council_vote_c = yes }
            }
        }
    }
}

# Tianzi-scoped synchronous safety net. AI writes are batched so 60+ ballots
# require one authoritative recount rather than one country scan per ballot.
zhx_fill_missing_tianxia_debate_ai_votes = {
    event_target:zhx_tianzi = { set_country_flag = zhx_council_batching_ai_votes }
    every_country = {
        limit = {
            ai = yes
            zhx_is_tianxia_polity = yes
            NOT = {
                OR = {
                    has_country_flag = zhx_council_vote_a
                    has_country_flag = zhx_council_vote_b
                    has_country_flag = zhx_council_vote_c
                }
            }
        }
        zhx_choose_tianxia_council_ai_vote = yes
    }
    event_target:zhx_tianzi = {
        clr_country_flag = zhx_council_batching_ai_votes
        zhx_recount_tianxia_council_ballot = yes
    }
}

zhx_open_tianxia_council = {
    clr_country_flag = zhx_council_result_ready
    zhx_clear_tianxia_council_phase = yes
    set_country_flag = zhx_council_phase_preparing
    zhx_clear_tianxia_council_results = yes
    zhx_reset_tianxia_council_ballot = yes
    clr_country_flag = zhx_council_phase_preparing
    set_country_flag = zhx_council_phase_ballot_open
    set_country_flag = zhx_council_deadline_scheduled
    zhx_recount_tianxia_council_ballot = yes

    if = {
        limit = { has_country_flag = zhx_council_kind_ritual_breakdown }
        every_country = {
            limit = { zhx_is_tianxia_member = yes }
            country_event = { id = zhx_system.21 }
        }
    }
    else_if = {
        limit = {
            has_country_flag = zhx_council_kind_debate
            zhx_council_has_valid_candidates = yes
        }
        every_country = {
            limit = { ai = yes zhx_is_tianxia_polity = yes }
            set_country_flag = zhx_council_ai_review_pending
            country_event = { id = zhx_debate.12 days = 182 random = 30 }
        }
        # First votes are synchronous: the public panel is complete as soon as
        # it opens, while a day-61 sweep still catches countries joining later.
        zhx_fill_missing_tianxia_debate_ai_votes = yes
        every_country = {
            limit = { ai = no zhx_is_tianxia_polity = yes }
            country_event = { id = zhx_debate.10 }
        }
        country_event = { id = zhx_debate.14 days = 61 }
    }
    country_event = { id = zhx_system.23 days = 365 }
}

zhx_begin_ritual_breakdown_council = {
    if = {
        limit = {
            NOT = { zhx_tianxia_council_is_busy = yes }
            NOT = { has_country_modifier = zhx_ritual_breakdown_incident_cooldown }
        }
        zhx_clear_tianxia_council_kind = yes
        zhx_clear_tianxia_council_candidates = yes
        clr_country_flag = zhx_council_pending_ritual_breakdown
        set_country_flag = zhx_council_kind_ritual_breakdown
        zhx_open_tianxia_council = yes
    }
}

zhx_begin_tianxia_debate_council = {
    if = {
        limit = {
            NOT = { zhx_tianxia_council_is_busy = yes }
            zhx_council_has_valid_candidates = yes
        }
        zhx_clear_tianxia_council_kind = yes
        set_country_flag = zhx_council_kind_debate
        zhx_open_tianxia_council = yes
    }
}

zhx_resolve_tianxia_council = {
    if = {
        limit = { has_country_flag = zhx_council_kind_debate }
        zhx_fill_missing_tianxia_debate_ai_votes = yes
    }
    clr_country_flag = zhx_council_deadline_scheduled
    zhx_clear_tianxia_council_phase = yes
    set_country_flag = zhx_council_phase_resolving
    set_country_flag = zhx_council_result_ready
    zhx_recount_tianxia_council_ballot = yes
}

zhx_finish_tianxia_council = {
    clr_country_flag = zhx_council_result_ready
    zhx_reset_tianxia_council_ballot = yes
    zhx_clear_tianxia_council_phase = yes
    zhx_clear_tianxia_council_kind = yes
    zhx_clear_tianxia_council_candidates = yes
    zhx_clear_tianxia_council_results = yes
    clr_country_flag = zhx_council_deadline_scheduled

    if = {
        limit = {
            has_country_flag = zhx_council_pending_ritual_breakdown
            NOT = { has_country_modifier = zhx_ritual_breakdown_incident_cooldown }
        }
        clr_country_flag = zhx_council_pending_ritual_breakdown
        country_event = { id = zhx_system.20 days = 1 }
    }
    else_if = {
        limit = {
            has_country_flag = zhx_council_pending_ritual_breakdown
            has_country_modifier = zhx_ritual_breakdown_incident_cooldown
        }
        clr_country_flag = zhx_council_pending_ritual_breakdown
    }
}

zhx_initialise_tianxia_council = {
    zhx_reset_tianxia_council_ballot = yes
    zhx_clear_tianxia_council_phase = yes
    zhx_clear_tianxia_council_kind = yes
    zhx_clear_tianxia_council_candidates = yes
    zhx_clear_tianxia_council_results = yes
    clr_country_flag = zhx_council_result_ready
    clr_country_flag = zhx_council_deadline_scheduled
    clr_country_flag = zhx_council_pending_ritual_breakdown
    set_global_flag = zhx_council_initialised_v1
}

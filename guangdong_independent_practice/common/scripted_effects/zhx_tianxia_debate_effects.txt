# Read-only, on-demand Six Schools ledger. Nothing is cached on a pulse.
zhx_refresh_tianxia_proposal_ledger = {
    set_variable = { which = zhx_tianxia_school_ru_total_count value = 0 }
    set_variable = { which = zhx_tianxia_school_fa_total_count value = 0 }
    set_variable = { which = zhx_tianxia_school_mo_total_count value = 0 }
    set_variable = { which = zhx_tianxia_school_dao_total_count value = 0 }
    set_variable = { which = zhx_tianxia_school_bing_total_count value = 0 }
    set_variable = { which = zhx_tianxia_school_zongheng_total_count value = 0 }
    set_variable = { which = zhx_tianxia_proposal_ru_count value = 0 }
    set_variable = { which = zhx_tianxia_proposal_fa_count value = 0 }
    set_variable = { which = zhx_tianxia_proposal_mo_count value = 0 }
    set_variable = { which = zhx_tianxia_proposal_dao_count value = 0 }
    set_variable = { which = zhx_tianxia_proposal_bing_count value = 0 }
    set_variable = { which = zhx_tianxia_proposal_zongheng_count value = 0 }

    every_country = {
        limit = {
            exists = yes
            zhx_is_tianxia_polity = yes
            zhx_is_lijiao_country = yes
        }
        if = {
            limit = { has_country_flag = zhx_doctrine_ru }
            ROOT = { change_variable = { which = zhx_tianxia_school_ru_total_count value = 1 } }
            if = {
                limit = { check_variable = { which = zhx_doctrine_practice value = 70 } }
                ROOT = { change_variable = { which = zhx_tianxia_proposal_ru_count value = 1 } }
            }
        }
        if = {
            limit = { has_country_flag = zhx_doctrine_fa }
            ROOT = { change_variable = { which = zhx_tianxia_school_fa_total_count value = 1 } }
            if = {
                limit = { check_variable = { which = zhx_doctrine_practice value = 70 } }
                ROOT = { change_variable = { which = zhx_tianxia_proposal_fa_count value = 1 } }
            }
        }
        if = {
            limit = { has_country_flag = zhx_doctrine_mo }
            ROOT = { change_variable = { which = zhx_tianxia_school_mo_total_count value = 1 } }
            if = {
                limit = { check_variable = { which = zhx_doctrine_practice value = 70 } }
                ROOT = { change_variable = { which = zhx_tianxia_proposal_mo_count value = 1 } }
            }

        }

# A or B must strictly lead both alternatives. Every tie for first becomes C,
# the positive settlement that multiple schools remain legitimate.
zhx_resolve_tianxia_debate = {
    zhx_clear_tianxia_council_results = yes
    if = {
        limit = {
            check_variable = { which = zhx_council_a_vs_b value = 1 }
            check_variable = { which = zhx_council_a_vs_c value = 1 }
        }
        set_country_flag = zhx_council_result_a
        zhx_apply_council_candidate_a_as_orthodoxy = yes
        change_variable = { which = zhx_ritual_authority value = 5 }
    }
    else_if = {
        limit = {
            check_variable = { which = zhx_council_b_vs_a value = 1 }
            check_variable = { which = zhx_council_b_vs_c value = 1 }
        }
        set_country_flag = zhx_council_result_b
        zhx_apply_council_candidate_b_as_orthodoxy = yes
        change_variable = { which = zhx_ritual_authority value = 5 }
    }
    else = {
        set_country_flag = zhx_council_result_c
        zhx_set_tianxia_orthodoxy_plural = yes
        change_variable = { which = zhx_ritual_authority value = 2 }
    }
    zhx_clamp_ritual_authority = yes
    zhx_refresh_ritual_order = yes
}

zhx_reward_tianxia_debate_alignment = {
    every_country = {
        limit = {
            exists = yes
            zhx_is_tianxia_member = yes
            OR = {
                AND = { ROOT = { has_country_flag = zhx_council_result_a } has_country_flag = zhx_council_vote_a }
                AND = { ROOT = { has_country_flag = zhx_council_result_b } has_country_flag = zhx_council_vote_b }
            }
        }
        change_variable = { which = zhx_merit value = 2 }
        zhx_clamp_merit = yes
    }
    every_country = {
        limit = {
            exists = yes
            zhx_is_tianxia_member = yes
            ROOT = { has_country_flag = zhx_council_result_c }
            has_country_flag = zhx_council_vote_c
        }
        change_variable = { which = zhx_merit value = 1 }
        zhx_clamp_merit = yes
    }
}

zhx_finish_tianxia_debate = {
    zhx_reward_tianxia_debate_alignment = yes
    add_country_modifier = {
        name = zhx_tianxia_debate_cooldown
        duration = 5475
    }
    zhx_finish_tianxia_council = yes
}

    set_variable = { which = zhx_tianxia_school_dao_total_count value = 0 }

            }
        }
        if = {
            limit = { has_country_flag = zhx_doctrine_fa }
            ROOT = { change_variable = { which = zhx_tianxia_school_fa_total_count value = 1 } }
            if = {
                limit = { check_variable = { which = zhx_doctrine_practice value = 70 } }
                ROOT = { change_variable = { which = zhx_tianxia_proposal_fa_count value = 1 } }
            }
        }
        if = {
            limit = { has_country_flag = zhx_doctrine_mo }
            ROOT = { change_variable = { which = zhx_tianxia_school_mo_total_count value = 1 } }
            if = {
                limit = { check_variable = { which = zhx_doctrine_practice value = 70 } }
                ROOT = { change_variable = { which = zhx_tianxia_proposal_mo_count value = 1 } }

# With exactly two live petitions, fixed school order fills A then B without
# asking the Tianzi a redundant question.
zhx_auto_select_tianxia_debate_candidates = {
    zhx_clear_tianxia_council_candidates = yes
    if = { limit = { has_country_modifier = zhx_debate_petition_ru } zhx_set_council_candidate_a_ru = yes }
    else_if = { limit = { has_country_modifier = zhx_debate_petition_fa } zhx_set_council_candidate_a_fa = yes }
    else_if = { limit = { has_country_modifier = zhx_debate_petition_mo } zhx_set_council_candidate_a_mo = yes }
    else_if = { limit = { has_country_modifier = zhx_debate_petition_dao } zhx_set_council_candidate_a_dao = yes }
    else_if = { limit = { has_country_modifier = zhx_debate_petition_bing } zhx_set_council_candidate_a_bing = yes }
    else_if = { limit = { has_country_modifier = zhx_debate_petition_zongheng } zhx_set_council_candidate_a_zongheng = yes }

    if = { limit = { has_country_modifier = zhx_debate_petition_ru NOT = { has_country_flag = zhx_council_candidate_a_ru } } zhx_set_council_candidate_b_ru = yes }
    else_if = { limit = { has_country_modifier = zhx_debate_petition_fa NOT = { has_country_flag = zhx_council_candidate_a_fa } } zhx_set_council_candidate_b_fa = yes }
    else_if = { limit = { has_country_modifier = zhx_debate_petition_mo NOT = { has_country_flag = zhx_council_candidate_a_mo } } zhx_set_council_candidate_b_mo = yes }
    else_if = { limit = { has_country_modifier = zhx_debate_petition_dao NOT = { has_country_flag = zhx_council_candidate_a_dao } } zhx_set_council_candidate_b_dao = yes }
    else_if = { limit = { has_country_modifier = zhx_debate_petition_bing NOT = { has_country_flag = zhx_council_candidate_a_bing } } zhx_set_council_candidate_b_bing = yes }
    else_if = { limit = { has_country_modifier = zhx_debate_petition_zongheng NOT = { has_country_flag = zhx_council_candidate_a_zongheng } } zhx_set_council_candidate_b_zongheng = yes }
    zhx_start_selected_tianxia_debate = yes
}

    else_if = { limit = { has_country_flag = zhx_council_candidate_b_bing } zhx_set_tianxia_orthodoxy_bing = yes }
    else_if = { limit = { has_country_flag = zhx_council_candidate_b_zongheng } zhx_set_tianxia_orthodoxy_zongheng = yes }
}

# A or B must strictly lead both alternatives. Every tie for first becomes C,
# the positive settlement that multiple schools remain legitimate.
zhx_resolve_tianxia_debate = {
    zhx_clear_tianxia_council_results = yes
    if = {
        limit = {
            check_variable = { which = zhx_council_a_vs_b value = 1 }
            check_variable = { which = zhx_council_a_vs_c value = 1 }
        }
        set_country_flag = zhx_council_result_a
        zhx_apply_council_candidate_a_as_orthodoxy = yes
        change_variable = { which = zhx_ritual_authority value = 5 }
    }
    else_if = {
        limit = {
            check_variable = { which = zhx_council_b_vs_a value = 1 }
            check_variable = { which = zhx_council_b_vs_c value = 1 }
        }
        set_country_flag = zhx_council_result_b
        zhx_apply_council_candidate_b_as_orthodoxy = yes
        change_variable = { which = zhx_ritual_authority value = 5 }
    }
    else = {
        set_country_flag = zhx_council_result_c
        zhx_set_tianxia_orthodoxy_plural = yes
        change_variable = { which = zhx_ritual_authority value = 2 }
    }
    zhx_clamp_ritual_authority = yes
    zhx_refresh_ritual_order = yes
}

zhx_reward_tianxia_debate_alignment = {
    every_country = {
        limit = {
            exists = yes
            zhx_is_tianxia_member = yes
            OR = {
                AND = { ROOT = { has_country_flag = zhx_council_result_a } has_country_flag = zhx_council_vote_a }
                AND = { ROOT = { has_country_flag = zhx_council_result_b } has_country_flag = zhx_council_vote_b }
            }
        }
        change_variable = { which = zhx_merit value = 2 }
        zhx_clamp_merit = yes
    }
    every_country = {
        limit = {
            exists = yes
            zhx_is_tianxia_member = yes
            ROOT = { has_country_flag = zhx_council_result_c }
            has_country_flag = zhx_council_vote_c
        }
        change_variable = { which = zhx_merit value = 1 }
        zhx_clamp_merit = yes
    }
}

zhx_finish_tianxia_debate = {
    zhx_reward_tianxia_debate_alignment = yes
    add_country_modifier = {
        name = zhx_tianxia_orthodoxy_term
        duration = 5475
    }
    zhx_finish_tianxia_council = yes
}

gdd_toggle_reform_support_effect = {
    # Custom GUI buttons otherwise expand the complete variable rebuild into
    # their generated tooltip.  The localized tooltip already exposes the two
    # useful totals, so ballot mutation and cache maintenance stay hidden.
    hidden_effect = {
        if = {
            limit = { has_country_flag = $support_flag$ }
            clr_country_flag = $support_flag$
        }
        else = {
            set_country_flag = $support_flag$
        }
        # Human overlords change their own ballot directly.  Refresh their AI
        # subjects immediately so the subjects' derived ballots do not wait
        # for the next quarterly pass.
        every_subject_country = {
            limit = {
                exists = yes
                ai = yes
                zhx_is_tianxia_member = yes
            }
            gdd_ai_evaluate_all_reform_votes_effect = yes
        }
        CZH = { gdd_recalculate_all_reform_votes_effect = yes }
    }
}

# Shared by startup and quarterly hidden events.  Keeping this in one scripted
# effect prevents the two schedules from drifting apart.
gdd_refresh_all_ai_reform_votes_effect = {
    gdd_clear_passed_reform_support_effect = yes
    # First evaluate independent members and subjects of powers outside the
    # Zhou realm.  Their ballots are the inputs used by member subjects below.
    every_country = {
        limit = {
            exists = yes
            ai = yes
            zhx_is_tianxia_member = yes
            NOT = {
                AND = {
                    is_subject = yes
                    overlord = { zhx_is_tianxia_polity = yes }
                }
            }
        }
        gdd_ai_evaluate_all_reform_votes_effect = yes
    }
    # Then evaluate subjects whose overlord is also part of the Zhou realm.
    # This makes the overlord's freshly calculated ballot a weighted input,
    # closely following the HRE model without replacing the subject's own AI.
    every_country = {
        limit = {
            exists = yes
            ai = yes
            zhx_is_tianxia_member = yes
            is_subject = yes
            overlord = { zhx_is_tianxia_polity = yes }
        }
        gdd_ai_evaluate_all_reform_votes_effect = yes
    }
    gdd_recalculate_all_reform_votes_effect = yes
}

# AI sponsor score.  At negative authority the current principal feudatory is
# the political sponsor; at zero or positive authority the Emperor is treated
# as the institutional sponsor even though reforms remain blocked at -25..25.
gdd_prepare_ai_reform_sponsor_score_effect = {
    set_variable = { which = gdd_ai_reform_sponsor_score value = 0 }
    if = {
        limit = {
            CZH = {
                NOT = { check_variable = { which = gdd_authority_integer_display value = -25 } }
            }
            has_saved_global_event_target = gdd_principal_vassal
        }
        if = {
            limit = { tag = event_target:gdd_principal_vassal }
            set_variable = { which = gdd_ai_reform_sponsor_score value = 6 }
        }
        else = {
            if = {
                limit = { has_opinion = { who = event_target:gdd_principal_vassal value = 100 } }
                change_variable = { which = gdd_ai_reform_sponsor_score value = 3 }
            }
            else_if = {
                limit = { has_opinion = { who = event_target:gdd_principal_vassal value = 50 } }
                change_variable = { which = gdd_ai_reform_sponsor_score value = 2 }
            }
            else_if = {
                limit = { has_opinion = { who = event_target:gdd_principal_vassal value = 0 } }
                change_variable = { which = gdd_ai_reform_sponsor_score value = 1 }
            }
            else = {
                change_variable = { which = gdd_ai_reform_sponsor_score value = -2 }
            }
            if = {
                limit = { alliance_with = event_target:gdd_principal_vassal }
                change_variable = { which = gdd_ai_reform_sponsor_score value = 2 }
            }
            if = {
                limit = { is_subject_of = event_target:gdd_principal_vassal }
                change_variable = { which = gdd_ai_reform_sponsor_score value = 2 }
            }
            if = {
                limit = {
                    OR = {
                        is_rival = event_target:gdd_principal_vassal
                        war_with = event_target:gdd_principal_vassal
                    }
                }
                set_variable = { which = gdd_ai_reform_sponsor_score value = -100 }
            }
        }
    }
    else = {
        if = {
            limit = { has_opinion = { who = event_target:EmperorOfChina value = 100 } }
            change_variable = { which = gdd_ai_reform_sponsor_score value = 3 }
        }
        else_if = {
            limit = { has_opinion = { who = event_target:EmperorOfChina value = 50 } }
            change_variable = { which = gdd_ai_reform_sponsor_score value = 2 }
        }
        else_if = {
            limit = { has_opinion = { who = event_target:EmperorOfChina value = 0 } }
            change_variable = { which = gdd_ai_reform_sponsor_score value = 1 }
        }
        else = {
            change_variable = { which = gdd_ai_reform_sponsor_score value = -2 }
        }
        if = {
            limit = { alliance_with = event_target:EmperorOfChina }
            change_variable = { which = gdd_ai_reform_sponsor_score value = 2 }
        }
        if = {
            limit = { is_subject_of = event_target:EmperorOfChina }
            change_variable = { which = gdd_ai_reform_sponsor_score value = 2 }
        }
        if = {
            limit = {
                OR = {
                    is_rival = event_target:EmperorOfChina
                    war_with = event_target:EmperorOfChina
                }
            }
            set_variable = { which = gdd_ai_reform_sponsor_score value = -100 }
        }
    }
}

gdd_begin_ai_reform_vote_effect = {
    set_variable = {
        which = gdd_ai_reform_vote_score
        which = gdd_ai_reform_sponsor_score
    }
}

gdd_apply_ai_reform_vote_effect = {
    # HRE-style subject influence: a member subject keeps its own score, while
    # a Zhou overlord's current ballot contributes a strong but non-absolute
    # +2/-2.  A foreign overlord cannot cast a Zhou ballot and instead gives a
    # small institutional-legitimacy penalty.
    if = {
        limit = {
            is_subject = yes
            overlord = { zhx_is_tianxia_polity = yes }
        }
        if = {
            limit = {
                overlord = { has_country_flag = $support_flag$ }
            }
            change_variable = { which = gdd_ai_reform_vote_score value = 2 }
        }
        else = {
            change_variable = { which = gdd_ai_reform_vote_score value = -2 }
        }
    }
    else_if = {
        limit = { is_subject = yes }
        change_variable = { which = gdd_ai_reform_vote_score value = -1 }
    }
    if = {
        limit = { check_variable = { which = gdd_ai_reform_vote_score value = 3 } }
        set_country_flag = $support_flag$
    }
    else = {
        clr_country_flag = $support_flag$
    }
}

    every_country = {
        limit = {
            has_country_flag = zhx_tianzi

}

# 只移除五个互斥的礼乐档位修正；已经通过的盟约等其他周天下修正会被保留。

    remove_country_modifier = zhx_ritual_order_ordered

# 初始化之后新加入周天下的国家也通过这个公共效果同步修正。
zhx_apply_current_ritual_order_modifier = {
    zhx_clear_ritual_order_modifier = yes

    if = {

            }

            name = zhx_ritual_order_ordered
            duration = -1
        }
    }
    else_if = {
        limit = {

                    value = 20

                        which = zhx_ritual_authority
                        value = 60

            }

                    value = -20

                NOT = {

# 统计当前仍在进行的诸侯内战，并把结果缓存到天子变量中。
# 外层国家是进攻方战争领袖，内层国家是防守方战争领袖；双方都必须是诸侯。
# 因此天子主动攻击诸侯、诸侯对外战争以及仅作为盟友参战都不会被误算为诸侯内战。
zhx_refresh_internal_war_count = {
    set_variable = {
        which = zhx_internal_war_count_cache
        value = 0
    }

    every_country = {
        limit = {
            exists = yes
            zhx_is_tianxia_member = yes
            is_at_war = yes
        }
        every_country = {
            limit = {
                exists = yes
                zhx_is_tianxia_member = yes
                defensive_war_with = PREV
                is_in_war = {
                    attacker_leader = PREV
                    defender_leader = THIS
                }
            }
            event_target:zhx_tianzi = {
                change_variable = {
                    which = zhx_internal_war_count_cache
                    value = 1
                }
            }
        }
    }
}

# 周天下国家作为战争胜方击败周天下以外的战争领袖时，当前天下显学改变
# 这场胜利的国际评价。ROOT 仍是胜方战争领袖，当前作用域则是天子锚点。

        else_if = {
            limit = { has_country_flag = zhx_tianxia_orthodoxy_fa }

            ROOT = { if = { limit = { zhx_is_tianxia_member = yes } change_variable = { which = zhx_merit value = 3 } zhx_clamp_merit = yes } }
        }
        else_if = { limit = { has_country_flag = zhx_tianxia_orthodoxy_mo } change_variable = { which = zhx_ritual_authority value = 1 } }
        else_if = {
            limit = { has_country_flag = zhx_tianxia_orthodoxy_dao }

            ROOT = { if = { limit = { zhx_is_tianxia_member = yes } change_variable = { which = zhx_merit value = 1 } zhx_clamp_merit = yes } }
        }
        else_if = {
            limit = { has_country_flag = zhx_tianxia_orthodoxy_bing }
            change_variable = { which = zhx_ritual_authority value = 5 }
            ROOT = { if = { limit = { zhx_is_tianxia_member = yes } change_variable = { which = zhx_merit value = 4 } zhx_clamp_merit = yes } }
        }
        else_if = {
            limit = { has_country_flag = zhx_tianxia_orthodoxy_zongheng }
            change_variable = { which = zhx_ritual_authority value = 3 }
            ROOT = { if = { limit = { zhx_is_tianxia_member = yes } change_variable = { which = zhx_merit value = 2 } zhx_clamp_merit = yes } }

# 天下事变与天下大辩的票盘、改票、重点和收尾均由
# zhx_council_effects.txt 的唯一公共内核负责。

# 礼乐值的低频年度结算，只在天子作用域中每年执行一次。
# 第一版使用少量、整数且便于玩家理解的因素；专属战争目标暂不产生额外影响。
zhx_yearly_ritual_authority_tick = {
    # 时间流逝会缓慢恢复礼制：固定每年 +1。
    change_variable = { which = zhx_ritual_authority value = 1 }

    # 每年重算一次内战数量，不建立每日或每月轮询。
    zhx_refresh_internal_war_count = yes

    # 第一版“天下无战事”包括对外战争：所有周天下国家都未参战时额外 +2。
    if = {
        limit = { zhx_tianxia_is_at_peace = yes }
        change_variable = { which = zhx_ritual_authority value = 2 }
    }

    # 诸侯内战按规模分档：1—2场 -1，3—5场 -2，6场及以上 -3。
    # Clausewitz 的 check_variable 是“至少达到该值”，所以必须从最高档向下判断。
    if = {
        limit = {
            check_variable = {
                which = zhx_internal_war_count_cache
                value = 6
            }
        }
        change_variable = { which = zhx_ritual_authority value = -3 }
    }
    else_if = {
        limit = {
            check_variable = {
                which = zhx_internal_war_count_cache
                value = 3
            }
        }
        change_variable = { which = zhx_ritual_authority value = -2 }
    }
    else_if = {
        limit = {
            check_variable = {
                which = zhx_internal_war_count_cache
                value = 1
            }
        }
        change_variable = { which = zhx_ritual_authority value = -1 }
    }

    # The current fifteen-year international orthodoxy changes how the same
    # world state is judged. These are international norms, not domestic buffs.
    if = {
        limit = {
            has_country_modifier = zhx_tianxia_orthodoxy_term
            zhx_tianxia_has_active_orthodoxy = yes
        }
        if = {
            limit = { has_country_flag = zhx_tianxia_orthodoxy_ru }
            if = { limit = { zhx_tianxia_is_at_peace = yes } change_variable = { which = zhx_ritual_authority value = 1 } }
            if = { limit = { check_variable = { which = zhx_internal_war_count_cache value = 1 } } change_variable = { which = zhx_ritual_authority value = -1 } }
        }
        else_if = {
            limit = { has_country_flag = zhx_tianxia_orthodoxy_fa }
            if = {
                limit = { NOT = { check_variable = { which = zhx_internal_war_count_cache value = 1 } } }
                change_variable = { which = zhx_ritual_authority value = 1 }
            }
            else_if = {
                limit = { check_variable = { which = zhx_internal_war_count_cache value = 3 } }
                change_variable = { which = zhx_ritual_authority value = -2 }
            }
        }
        else_if = {
            limit = { has_country_flag = zhx_tianxia_orthodoxy_mo }
            if = { limit = { zhx_tianxia_is_at_peace = yes } change_variable = { which = zhx_ritual_authority value = 2 } }
            if = { limit = { check_variable = { which = zhx_internal_war_count_cache value = 1 } } change_variable = { which = zhx_ritual_authority value = -2 } }
        }
        else_if = {
            limit = { has_country_flag = zhx_tianxia_orthodoxy_dao }
            if = { limit = { zhx_tianxia_is_at_peace = yes } change_variable = { which = zhx_ritual_authority value = 2 } }
            if = { limit = { check_variable = { which = zhx_internal_war_count_cache value = 1 } } change_variable = { which = zhx_ritual_authority value = -1 } }
        }
        else_if = {
            limit = { has_country_flag = zhx_tianxia_orthodoxy_bing }
            if = { limit = { NOT = { check_variable = { which = zhx_internal_war_count_cache value = 1 } } } change_variable = { which = zhx_ritual_authority value = 1 } }
            if = { limit = { check_variable = { which = zhx_internal_war_count_cache value = 3 } } change_variable = { which = zhx_ritual_authority value = -1 } }
        }
        else_if = {
            limit = { has_country_flag = zhx_tianxia_orthodoxy_zongheng }
            if = { limit = { NOT = { check_variable = { which = zhx_internal_war_count_cache value = 1 } } } change_variable = { which = zhx_ritual_authority value = 1 } }
            if = { limit = { check_variable = { which = zhx_internal_war_count_cache value = 3 } } change_variable = { which = zhx_ritual_authority value = -1 } }
        }
    }

    # Clausewitz 的数值触发器是下限判断：stability = 1 表示稳定度 >= 1。
    # NOT = { stability = 0 } 因而表示稳定度 < 0。
    if = {
        limit = { stability = 1 }
        change_variable = { which = zhx_ritual_authority value = 1 }
    }
    else_if = {
        limit = { NOT = { stability = 0 } }
        change_variable = { which = zhx_ritual_authority value = -2 }
    }

    # 同理，prestige = 50 表示威望 >= 50；NOT prestige = -25 表示威望 < -25。
    if = {
        limit = { prestige = 50 }
        change_variable = { which = zhx_ritual_authority value = 1 }
    }
    else_if = {
        limit = { NOT = { prestige = -25 } }
        change_variable = { which = zhx_ritual_authority value = -1 }
    }

    # 天子的厌战度达到 5 时，每年额外 -1。
    if = {
        limit = { war_exhaustion = 5 }
        change_variable = { which = zhx_ritual_authority value = -1 }
    }

    # 进入 capital_scope 后当前作用域变为首都省份，但 ROOT 仍指向外层的天子国家。
    # 首都不受天子本人控制时，每年额外 -5。
    if = {
        limit = {
            capital_scope = {
                NOT = { controlled_by = ROOT }
            }
        }
        change_variable = { which = zhx_ritual_authority value = -5 }
    }

    # 先把礼乐值钳制在 -100..100，再根据最终数值刷新档位旗标、成员修正和 GUI 状态。
    zhx_clamp_ritual_authority = yes
    zhx_refresh_ritual_order = yes

    # 礼乐值跌破 -50 时登记“王命不行”。若另一项公议正在进行，事变
    # 进入待办而不是另开第二套票盘；当前公议结束后再启动。
    if = {
        limit = {
            NOT = {
                check_variable = {
                    which = zhx_ritual_authority
                    value = -50
                }
            }
            NOT = { has_country_modifier = zhx_ritual_breakdown_incident_cooldown }
        }
        if = {
            limit = {
                zhx_tianxia_council_is_busy = yes
                NOT = { has_country_flag = zhx_council_kind_ritual_breakdown }
            }
            set_country_flag = zhx_council_pending_ritual_breakdown
        }
        else_if = {
            limit = { NOT = { zhx_tianxia_council_is_busy = yes } }
            # Reserve the queue before showing the proposal event. The yearly
            # debate check runs later in the same pulse and must not overtake it.
            set_country_flag = zhx_council_pending_ritual_breakdown
            country_event = { id = zhx_system.20 }
        }
    }
}

# 把 ROOT 登记为正式诸侯。成员国家自身的 zhx_member 旗标是唯一真相；
# 天子保存的成员数量只是可随时重建的 GUI/显示缓存。
zhx_register_tianxia_member = {
    if = {
        limit = {
            exists = yes
            NOT = { has_country_flag = zhx_tianzi }
            NOT = { has_country_flag = zhx_member }
        }
        set_country_flag = zhx_member
        set_variable = {
            which = zhx_merit
            value = 0
        }
        event_target:zhx_tianzi = {
            change_variable = {
                which = zhx_member_count_cache
                value = 1
            }
        }
        zhx_apply_current_ritual_order_modifier = yes
        if = {
            limit = { has_global_flag = zhx_gui_roster_initialised }
            event_target:zhx_tianzi = {
                zhx_build_gui_roster = yes
            }
        }
    }
}

zhx_register_major_feudatory = {
    if = {
        limit = {
            exists = yes
            NOT = { has_country_flag = zhx_tianzi }
        }
        zhx_register_tianxia_member = yes
        set_country_flag = zhx_major_feudatory
    }
}

zhx_remove_tianxia_member = {
    if = {
        limit = {
            OR = {
                has_country_flag = zhx_council_vote_a
                has_country_flag = zhx_council_vote_b
                has_country_flag = zhx_council_vote_c
            }
        }
        zhx_clear_current_tianxia_council_vote = yes
        event_target:zhx_tianzi = {
            if = {
                limit = { has_country_flag = zhx_council_phase_ballot_open }
                zhx_recount_tianxia_council_ballot = yes
            }
        }
    }
    if = {
        limit = {
            has_country_flag = zhx_member
        }
        event_target:zhx_tianzi = {
            change_variable = {
                which = zhx_member_count_cache
                value = -1
            }
        }
    }
    clr_country_flag = zhx_member
    clr_country_flag = zhx_major_feudatory
    set_variable = {
        which = zhx_merit
        value = 0
    }
    zhx_clear_ritual_order_modifier = yes
    remove_country_modifier = zhx_reaffirmed_covenant_modifier
    if = {
        limit = { has_global_flag = zhx_gui_roster_initialised }
        event_target:zhx_tianzi = {
            zhx_build_gui_roster = yes
        }
    }
}

# 在存档迁移、控制台改动或未来天子更替后修复成员数量缓存。
# 这是修复工具，不应作为每月或每日循环使用。
zhx_rebuild_tianxia_member_cache = {
    event_target:zhx_tianzi = {
        set_variable = {
            which = zhx_member_count_cache
            value = 0
        }
    }
    every_country = {
        limit = {
            has_country_flag = zhx_member
        }
        event_target:zhx_tianzi = {
            change_variable = {
                which = zhx_member_count_cache
                value = 1
            }
        }
    }
}

# 将早期只有身份登记的旧存档迁移到 v3：保留既有成员旗标，建立礼乐状态，
# 最后只重建可派生的计数缓存与国家修正。
zhx_migrate_tianxia_system_v2_to_v3 = {
    CZH = {
        zhx_set_tianzi = yes
        set_variable = {
            which = zhx_ritual_authority
            value = 50
        }
        set_variable = { which = zhx_council_vote_a_count value = 0 }
        set_variable = { which = zhx_council_vote_b_count value = 0 }
        set_variable = { which = zhx_council_vote_c_count value = 0 }
    }

    every_country = {
        limit = { has_country_flag = zhx_member }
        set_variable = {
            which = zhx_merit
            value = 0
        }
    }

    zhx_rebuild_tianxia_member_cache = yes
    CZH = { zhx_refresh_ritual_order = yes }
    set_global_flag = zhx_system_initialised_v3

    every_country = {
        limit = {
            exists = yes
            ai = no
        }
        clr_country_flag = zhx_startup_overview_seen
        country_event = { id = zhx_system.2 }
    }
}

# 将 v3 旧存档的 0..100 礼乐值线性迁移到对称的 -100..100 轴。
# 旧边界 20/40/60/80 会准确对应到新边界 -60/-20/20/60。
zhx_migrate_tianxia_system_v3_to_v4 = {
    CZH = {
        multiply_variable = {
            which = zhx_ritual_authority
            value = 2
        }
        change_variable = {
            which = zhx_ritual_authority
            value = -100
        }
        clr_country_flag = zhx_ritual_tier_ordered
        clr_country_flag = zhx_ritual_tier_stable
        clr_country_flag = zhx_ritual_tier_balanced
        clr_country_flag = zhx_ritual_tier_fractured
        clr_country_flag = zhx_ritual_tier_collapsed
        zhx_clamp_ritual_authority = yes
        zhx_refresh_ritual_order = yes
    }
    set_global_flag = zhx_system_initialised_v4
}

# 把以延安为首都的义渠（YOU）补入已有存档，而不从头重建整个成员表。
# 公共登记效果会同时刷新成员计数、礼乐修正和 GUI 盾徽名册。
zhx_migrate_tianxia_system_v4_to_v5 = {
    if = {
        limit = {
            YOU = {
                exists = yes
                NOT = { has_country_flag = zhx_tianzi }
                NOT = { has_country_flag = zhx_member }
            }
        }
        YOU = { zhx_register_tianxia_member = yes }
    }
    set_global_flag = zhx_system_initialised_v5
}

# v6 establishes the single public-council kernel. This project targets fresh
# campaigns, so no parallel legacy ballot is retained.
zhx_migrate_tianxia_system_v5_to_v6 = {
    CZH = {
        remove_country_modifier = zhx_ritual_breakdown_incident_cooldown
        zhx_initialise_tianxia_council = yes
    }
    set_global_flag = zhx_system_initialised_v6
}

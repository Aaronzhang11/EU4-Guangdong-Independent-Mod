# 将 ROOT 设为唯一的天子，并刷新供后续规则与 GUI 使用的全局事件目标。
# 天子身份只由周天下系统管理，刻意不依赖任何属国类型。
zhx_set_tianzi = {
    every_country = {
        limit = {
            has_country_flag = zhx_tianzi
        }
        clr_country_flag = zhx_tianzi
    }
    # If a future emperor was previously a normal member, remove it through
    # the common exit path first so the compact GUI roster and count cache do
    # not retain its former slot.
    if = {
        limit = { has_country_flag = zhx_member }
        zhx_remove_tianxia_member = yes
    }
    clr_country_flag = zhx_member
    clr_country_flag = zhx_major_feudatory
    clr_country_flag = zhx_ritual_tier_ordered
    clr_country_flag = zhx_ritual_tier_stable
    clr_country_flag = zhx_ritual_tier_balanced
    clr_country_flag = zhx_ritual_tier_fractured
    clr_country_flag = zhx_ritual_tier_collapsed
    set_country_flag = zhx_tianzi
    save_global_event_target_as = zhx_tianzi
}

# 统计当前仍在进行的诸侯内战，并把结果缓存到天子变量中。
# 外层国家是进攻方战争领袖，内层国家是防守方战争领袖；双方都必须是诸侯。
# 因此天子主动攻击诸侯、诸侯对外战争以及仅作为盟友参战都不会被误算为诸侯内战。
zhx_refresh_internal_war_count = {
    set_variable = {
        which = zhx_internal_war_count_cache
        value = 0
    }

    every_country = {
        limit = {
            exists = yes
            zhx_is_tianxia_member = yes
            is_at_war = yes
        }
        every_country = {
            limit = {
                exists = yes
                zhx_is_tianxia_member = yes
                defensive_war_with = PREV
                is_in_war = {
                    attacker_leader = PREV
                    defender_leader = THIS
                }
            }
            event_target:zhx_tianzi = {
                change_variable = {
                    which = zhx_internal_war_count_cache
                    value = 1
                }
            }
        }
    }
}

# 礼乐值的低频年度结算，只在天子作用域中每年执行一次。
# 第一版使用少量、整数且便于玩家理解的因素；专属战争目标暂不产生额外影响。
zhx_yearly_ritual_authority_tick = {
    # 时间流逝会缓慢恢复礼制：固定每年 +1。
    change_variable = { which = zhx_ritual_authority value = 1 }

    # 每年重算一次内战数量，不建立每日或每月轮询。
    zhx_refresh_internal_war_count = yes

    # 第一版“天下无战事”包括对外战争：所有周天下国家都未参战时额外 +2。
    if = {
        limit = { zhx_tianxia_is_at_peace = yes }
        change_variable = { which = zhx_ritual_authority value = 2 }
    }

    # 诸侯内战按规模分档：1—2场 -1，3—5场 -2，6场及以上 -3。
    # Clausewitz 的 check_variable 是“至少达到该值”，所以必须从最高档向下判断。
    if = {
        limit = {
            check_variable = {
                which = zhx_internal_war_count_cache
                value = 6
            }
        }
        change_variable = { which = zhx_ritual_authority value = -3 }
    }
    else_if = {
        limit = {
            check_variable = {
                which = zhx_internal_war_count_cache
                value = 3
            }
        }
        change_variable = { which = zhx_ritual_authority value = -2 }
    }
    else_if = {
        limit = {
            check_variable = {
                which = zhx_internal_war_count_cache
                value = 1
            }
        }
        change_variable = { which = zhx_ritual_authority value = -1 }
    }

    # Clausewitz 的数值触发器是下限判断：stability = 1 表示稳定度 >= 1。
    # NOT = { stability = 0 } 因而表示稳定度 < 0。
    if = {
        limit = { stability = 1 }
        change_variable = { which = zhx_ritual_authority value = 1 }
    }
    else_if = {
        limit = { NOT = { stability = 0 } }
        change_variable = { which = zhx_ritual_authority value = -2 }
    }

    # 同理，prestige = 50 表示威望 >= 50；NOT prestige = -25 表示威望 < -25。
    if = {
        limit = { prestige = 50 }
        change_variable = { which = zhx_ritual_authority value = 1 }
    }
    else_if = {
        limit = { NOT = { prestige = -25 } }
        change_variable = { which = zhx_ritual_authority value = -1 }
    }

    # 天子的厌战度达到 5 时，每年额外 -1。
    if = {
        limit = { war_exhaustion = 5 }
        change_variable = { which = zhx_ritual_authority value = -1 }
    }

    # 进入 capital_scope 后当前作用域变为首都省份，但 ROOT 仍指向外层的天子国家。
    # 首都不受天子本人控制时，每年额外 -5。
    if = {
        limit = {
            capital_scope = {
                NOT = { controlled_by = ROOT }
            }
        }
        change_variable = { which = zhx_ritual_authority value = -5 }
    }

    # 先把礼乐值钳制在 -100..100，再根据最终数值刷新档位旗标、成员修正和 GUI 状态。
    zhx_clamp_ritual_authority = yes
    zhx_refresh_ritual_order = yes

    # 礼乐值跌破 -50，且当前没有正在进行或处于冷却期的同类事变时，触发“王命不行”。
    if = {
        limit = {
            NOT = {
                check_variable = {
                    which = zhx_ritual_authority
                    value = -50
                }
            }
            NOT = { has_country_flag = zhx_ritual_breakdown_incident_active }
            NOT = { has_country_modifier = zhx_ritual_breakdown_incident_cooldown }
        }
        country_event = { id = zhx_system.20 }
    }
}

# 把 ROOT 登记为正式诸侯。成员国家自身的 zhx_member 旗标是唯一真相；
# 天子保存的成员数量只是可随时重建的 GUI/显示缓存。
zhx_register_tianxia_member = {
    if = {
        limit = {
            exists = yes
            NOT = { has_country_flag = zhx_tianzi }
            NOT = { has_country_flag = zhx_member }
        }
        set_country_flag = zhx_member
        set_variable = {
            which = zhx_merit
            value = 0
        }
        event_target:zhx_tianzi = {
            change_variable = {
                which = zhx_member_count_cache
                value = 1
            }
        }
        zhx_apply_current_ritual_order_modifier = yes
        if = {
            limit = { has_global_flag = zhx_gui_roster_initialised }
            event_target:zhx_tianzi = {
                zhx_build_gui_roster = yes
            }
        }
    }
}

zhx_register_major_feudatory = {
    if = {
        limit = {
            exists = yes
            NOT = { has_country_flag = zhx_tianzi }
        }
        zhx_register_tianxia_member = yes
        set_country_flag = zhx_major_feudatory
    }
}

    if = {
        limit = {
            OR = {
                has_country_flag = zhx_incident_vote_support
                has_country_flag = zhx_incident_vote_oppose
                has_country_flag = zhx_incident_vote_abstain
            }
        }
        zhx_clear_current_ritual_breakdown_vote = yes
    }
    if = {
        limit = {
            has_country_flag = zhx_member
        }
        event_target:zhx_tianzi = {
            change_variable = {
                which = zhx_member_count_cache
                value = -1
            }
        }
    }
    clr_country_flag = zhx_member
    clr_country_flag = zhx_major_feudatory
    clr_country_flag = zhx_incident_supporter
    set_variable = {
        which = zhx_merit
        value = 0
    }
    zhx_clear_ritual_order_modifier = yes
    remove_country_modifier = zhx_reaffirmed_covenant_modifier
    if = {
        limit = { has_global_flag = zhx_gui_roster_initialised }
        event_target:zhx_tianzi = {
            zhx_build_gui_roster = yes
        }
    }
}

# 在存档迁移、控制台改动或未来天子更替后修复成员数量缓存。
# 这是修复工具，不应作为每月或每日循环使用。
zhx_rebuild_tianxia_member_cache = {
    event_target:zhx_tianzi = {
        set_variable = {
            which = zhx_member_count_cache
            value = 0
        }

    every_country = {
        limit = {
            has_country_flag = zhx_member
        }
        event_target:zhx_tianzi = {
            change_variable = {
                which = zhx_member_count_cache
                value = 1
            }
        }
    }
}

    CZH = {

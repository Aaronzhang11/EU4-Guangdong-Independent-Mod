# bonuses are separated so a later change of dynasty automatically transfers
# the correct effects to the new holder of the Mandate.

    potential = { has_global_flag = gdd_proxy_reform_establish_gaituguiliu }
    trigger = { is_emperor_of_china = yes }
    meritocracy = 0.5
}

    potential = { has_global_flag = gdd_proxy_reform_establish_gaituguiliu }

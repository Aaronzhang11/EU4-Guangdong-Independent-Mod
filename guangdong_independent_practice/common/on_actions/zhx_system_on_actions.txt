on_startup = {
    events = {
        zhx_system.1
        zhx_opening_school.1
        zhx_debate.1
        gdd_celestial_reform_vote.1
        # Fresh 1444 custom nations are not part of the fixed 65-tag school
        # catalogue. The shared lifecycle gives eligible Ritual Teaching
        # designs the ordinary six-school foundation route and rejects an
        # ineligible hollow faith.
        zhx_doctrine.92
    }
    if = {
        limit = {
            ai = no
            zhx_is_lijiao_country = yes
            zhx_has_doctrine = yes
        }
        zhx_prepare_doctrine_ledger = yes
    }
}

# ROOT is the country after its religion has changed. The cleanup event clears
# doctrine gameplay state immediately and replaces same-group eastern mirrors
# with a transparent no-doctrine sentinel.
on_religion_change = {
    events = {
        zhx_doctrine.92
    }
}

# Ritual Teaching eligibility follows the country's primary culture rather
# than its tag alone. Reuse the same lifecycle event so culture promotion can
# neither strand a valid state without a school nor leave an ineligible state
# carrying the six-school mirror.
on_primary_culture_changed = {
    events = {
        zhx_doctrine.92
    }
}

# Released countries inherit their authored religion and culture without
# necessarily firing either change hook. Give a released Ritual Teaching state
# the same bounded eligibility/foundation pass.
on_country_released = {
    events = {
        zhx_doctrine.92
    }
}

# The religion-page hover ledger is presentation cache, not a second gameplay
# clock. Once per month, each human country with a live Ritual Teaching doctrine
# recomputes only its own bounded factor rows, raw total and threshold gaps.
# AI countries have no religion-page reader and need no cache refresh. There is
# no every_country loop and no province scan.
# Reform voting shares this monthly pulse, while its own event trigger limits
# the expensive refresh to January, April, July and October.
on_monthly_pulse = {
    events = {
        gdd_celestial_reform_vote.2
    }
    if = {
        limit = {
            ai = no
            zhx_is_lijiao_country = yes
            zhx_has_doctrine = yes
        }
        zhx_prepare_doctrine_ledger = yes
    }
}

# Subject status changes alter the new HRE-style overlord component of the AI
# score.  Refresh immediately when either side belongs to the Zhou realm;
# ROOT is the overlord and FROM is the subject in both on-actions.
on_dependency_gained = {
    if = {
        limit = {
            has_global_flag = gdd_celestial_reform_voting_initialised
            OR = {
                zhx_is_tianxia_polity = yes
                FROM = { zhx_is_tianxia_member = yes }
            }
        }
        CZH = { gdd_refresh_all_ai_reform_votes_effect = yes }
    }
}

on_dependency_lost = {
    if = {
        limit = {
            has_global_flag = gdd_celestial_reform_voting_initialised
            OR = {
                zhx_is_tianxia_polity = yes
                FROM = { zhx_is_tianxia_member = yes }
            }
        }
        CZH = { gdd_refresh_all_ai_reform_votes_effect = yes }
    }
}

# 年度脉冲继续处理原有周天下系统维护；改革投票已改为季度脉冲。
on_yearly_pulse = {

        zhx_system.10
        zhx_doctrine.90
        zhx_academy.90
        zhx_academy.300
        zhx_academy.301
        zhx_debate.90
    }
}

# Keep development-weighted reform votes exact after manual development,
# territorial transfers and province ownership changes.  The yearly pass is a
# final safety net for development changed by less common scripted effects.
on_adm_development = {
    owner = {
        if = {
            limit = { zhx_is_tianxia_member = yes }
            CZH = { gdd_recalculate_all_reform_votes_effect = yes }
        }
    }
}

on_province_owner_change = {
    # THIS is the transferred province and FROM is its former owner. Only the
    # twelve named academy provinces need reconciliation, so ordinary conquest
    # pays no academy scan cost. Both the new and old owner are refreshed at once.
    if = {
        limit = {
            OR = {
                has_province_modifier = zhx_academy_chongli
                has_province_modifier = zhx_academy_minglun
                has_province_modifier = zhx_academy_yifa
                has_province_modifier = zhx_academy_xunming
                has_province_modifier = zhx_academy_jianai
                has_province_modifier = zhx_academy_shangtong
                has_province_modifier = zhx_academy_xuantong
                has_province_modifier = zhx_academy_baopu
                has_province_modifier = zhx_academy_miaosuan
                has_province_modifier = zhx_academy_zhisheng
                has_province_modifier = zhx_academy_baihe
                has_province_modifier = zhx_academy_lianheng
            }
        }
        # Match the named academy against the former owner's corresponding
        # expelling flag. This must not depend on the temporary pressure marker:
        # the engine may discard that modifier during the transfer itself.
        zhx_academy_cancel_expulsion_on_owner_change = yes
        owner = {
            zhx_refresh_academy_country_effects = yes
        }
        if = {
            limit = { FROM = { exists = yes } }
            FROM = {
                zhx_refresh_academy_country_effects = yes
            }
        }
    }

    # Province transfers can change the development weight of both the new and
    # former Zhou members, so recalculate the realm vote once after the transfer.
    if = {
        limit = {
            OR = {
                owner = { zhx_is_tianxia_member = yes }
                FROM = { zhx_is_tianxia_member = yes }
            }
        }
        CZH = { gdd_recalculate_all_reform_votes_effect = yes }
    }
}
# 战争结束时 ROOT 是胜方战争领袖，FROM 是败方战争领袖。
# 只有周天下国家击败周天下以外的国家才结算一次；礼乐与盟功数值由当前天下显学决定。
on_war_ended = {
    if = {
        limit = {
            has_global_flag = zhx_system_initialised_v4
            zhx_is_tianxia_polity = yes
            FROM = {
                NOT = { zhx_is_tianxia_polity = yes }
            }
        }
        event_target:zhx_tianzi = {
            zhx_reward_external_war_victory = yes
        }
    }
}

# Reassign the authority effects immediately if the native Mandate changes
# hands.  The yearly refresh remains as a low-frequency safety net.
on_mandate_of_heaven_gained = {
    if = {
        limit = {
            has_global_flag = zhx_system_initialised_v8
            CZH = { exists = yes }
        }
        CZH = { gdd_refresh_authority_balance_effects = yes }
    }
}

on_mandate_of_heaven_lost = {
    if = {
        limit = {
            has_global_flag = zhx_system_initialised_v8
            CZH = { exists = yes }
        }
        CZH = { gdd_refresh_authority_balance_effects = yes }
    }
}

# Full annexation and diplomatic integration must use the same authoritative
# removal effect as an explicit expulsion. It clears the membership identity,
# fixes the count cache, and rebuilds both GUI rosters without leaving a hole.
# FROM is the country that is being removed in both on-actions.
on_integrate = {
    FROM = {
        if = {
            limit = { has_country_flag = zhx_member }
            zhx_remove_tianxia_member = yes
        }
    }
}

on_annexed = {
    FROM = {
        if = {
            limit = { has_country_flag = zhx_member }
            zhx_remove_tianxia_member = yes
        }
    }
}

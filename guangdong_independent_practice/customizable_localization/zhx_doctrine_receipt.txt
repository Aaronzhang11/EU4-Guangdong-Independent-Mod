# The compact religion-page hover is assembled from seven deterministic slots:
# one signed raw total, three positive rows, three negative rows, and an empty
# state. Cached flags are written by the same authoritative calculator as the
# yearly settlement, so only factors which actually contributed are rendered.
defined_text = {
    name = GetZhxDoctrinePracticeHoverTotal
    random = no

    text = {
        localisation_key = zhx_doctrine_practice_hover_total_positive
        trigger = {
            check_variable = {
                which = zhx_doctrine_ledger_estimated_delta
                value = 1
            }
        }
    }
    text = {
        localisation_key = zhx_doctrine_practice_hover_total_negative
        trigger = {
            NOT = {
                check_variable = {
                    which = zhx_doctrine_ledger_estimated_delta
                    value = 0
                }
            }
        }
    }
    text = {
        localisation_key = zhx_doctrine_practice_hover_total_zero
        trigger = { always = yes }
    }
}

# Live religion-page thought-tension tooltip. These accessors run only while
# the player hovers the panel; they read province modifiers and the academy
# lifecycle's mutually-exclusive country modifiers without adding cache state.

defined_text = {
    name = GetZhxThoughtTensionTier
    random = no

    text = {
        localisation_key = zhx_thought_tension_tier_heavy
        trigger = { has_country_modifier = zhx_academy_tension_heavy }
    }
    text = {
        localisation_key = zhx_thought_tension_tier_medium
        trigger = { has_country_modifier = zhx_academy_tension_medium }
    }
    text = {
        localisation_key = zhx_thought_tension_tier_mild
        trigger = { has_country_modifier = zhx_academy_tension_mild }
    }
    text = {
        localisation_key = zhx_thought_tension_tier_none
        trigger = { always = yes }
    }
}

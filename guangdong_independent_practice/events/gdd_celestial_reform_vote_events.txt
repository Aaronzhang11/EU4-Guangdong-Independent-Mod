# month is expressed as a closed one-month range.  This keeps full AI ballot
# evaluation to January, April, July and October instead of every month.

    id = gdd_celestial_reform_vote.2

            AND = { is_month = 3 NOT = { is_month = 4 } }

            AND = { is_month = 9 NOT = { is_month = 10 } }

        gdd_refresh_all_ai_reform_votes_effect = yes

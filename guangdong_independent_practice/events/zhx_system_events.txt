# 隐藏的初始化入口。固定由 CZH 承载，使后续代码拥有稳定的国家锚点，
# 不必为了寻找天子而进行每日或每月的全世界扫描。
country_event = {
    id = zhx_system.1
    title = zhx_system.1.t
    desc = zhx_system.1.d
    picture = DIPLOMACY_eventPicture
    hidden = yes
    is_triggered_only = yes

    trigger = {
        tag = CZH
        exists = yes
        OR = {
            NOT = { has_global_flag = zhx_system_initialised_v3 }
            NOT = { has_global_flag = zhx_system_initialised_v4 }
            NOT = { has_global_flag = zhx_system_initialised_v5 }
            NOT = { has_global_flag = zhx_system_initialised_v6 }
            NOT = { has_global_flag = zhx_gui_roster_initialised }
            NOT = { has_global_flag = gdd_eoc_member_roster_initialised }
        }
    }

    immediate = {
        if = {
            limit = {
                NOT = { has_global_flag = zhx_system_initialised_v3 }
            }
            if = {
                limit = { has_global_flag = zhx_system_initialised_v2 }
                zhx_migrate_tianxia_system_v2_to_v3 = yes
            }
            else = {
                zhx_initialise_tianxia_system = yes
            }
        }
        if = {
            limit = {
                NOT = { has_global_flag = zhx_system_initialised_v4 }
            }
            zhx_migrate_tianxia_system_v3_to_v4 = yes
        }
        if = {
            limit = {
                NOT = { has_global_flag = zhx_system_initialised_v5 }
            }
            zhx_migrate_tianxia_system_v4_to_v5 = yes
        }
        if = {
            limit = {
                NOT = { has_global_flag = zhx_system_initialised_v6 }
            }
            zhx_migrate_tianxia_system_v5_to_v6 = yes
        }
        if = {
            limit = {
                OR = {
                    NOT = { has_global_flag = zhx_gui_roster_initialised }
                    NOT = { has_global_flag = gdd_eoc_member_roster_initialised }
                }
            }
            zhx_build_gui_roster = yes
        }
    }

    option = {
        name = OK
    }
}

# 天下事变原型：复现原版帝国事变“先议论、后裁断”的生命周期，
# 但不尝试创建第二个硬编码帝国容器。延迟事件会被写入存档，
# 因而无需每日轮询，也不会因为读档而丢失一年后的结算。
country_event = {
    id = zhx_system.20
    title = zhx_system.20.t
    desc = zhx_system.20.d
    picture = DIPLOMACY_eventPicture
    major = yes
    is_triggered_only = yes

    trigger = {
        zhx_is_tianzi = yes
        NOT = { has_country_flag = zhx_ritual_breakdown_incident_active }
        NOT = { has_country_modifier = zhx_ritual_breakdown_incident_cooldown }
    }

    immediate = {
        hidden_effect = {
            # 新一轮公议开始前清理旧票，防止中途升级模组或异常结束留下脏状态。
            zhx_reset_ritual_breakdown_incident_votes = yes
            set_country_flag = zhx_ritual_breakdown_incident_active
        }
    }

    option = {
        name = zhx_system.20.a
        custom_tooltip = zhx_system.20.a.tt
        hidden_effect = {
            # 所有当前诸侯收到一张选票；AI 会立即表态，人类玩家可以在一年内改票。
            every_country = {
                limit = { zhx_is_tianxia_member = yes }
                country_event = { id = zhx_system.21 }
            }

            # 365 天后重新扫描仍然存在的成员及其最终票，不信任过程计数缓存。
            country_event = {
                id = zhx_system.22
                days = 365
            }
        }
        ai_chance = { factor = 1 }
    }
}

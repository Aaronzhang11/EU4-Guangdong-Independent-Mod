# 隐藏的初始化入口。固定由 CZH 承载，使后续代码拥有稳定的国家锚点，
# 不必为了寻找天子而进行每日或每月的全世界扫描。
country_event = {
    id = zhx_system.1
    title = zhx_system.1.t
    desc = zhx_system.1.d
    picture = DIPLOMACY_eventPicture
    hidden = yes
    is_triggered_only = yes

    trigger = {
        tag = CZH
        exists = yes
        OR = {
            NOT = { has_global_flag = zhx_system_initialised_v3 }
            NOT = { has_global_flag = zhx_system_initialised_v4 }
            NOT = { has_global_flag = zhx_system_initialised_v5 }
            NOT = { has_global_flag = zhx_system_initialised_v6 }
            NOT = { has_global_flag = zhx_gui_roster_initialised }
            NOT = { has_global_flag = gdd_eoc_member_roster_initialised }
        }
    }

    immediate = {
        if = {
            limit = {
                NOT = { has_global_flag = zhx_system_initialised_v3 }
            }
            if = {
                limit = { has_global_flag = zhx_system_initialised_v2 }
                zhx_migrate_tianxia_system_v2_to_v3 = yes
            }
            else = {
                zhx_initialise_tianxia_system = yes
            }
        }
        if = {
            limit = {
                NOT = { has_global_flag = zhx_system_initialised_v4 }
            }
            zhx_migrate_tianxia_system_v3_to_v4 = yes
        }
        if = {
            limit = {
                NOT = { has_global_flag = zhx_system_initialised_v5 }
            }
            zhx_migrate_tianxia_system_v4_to_v5 = yes
        }
        if = {
            limit = {
                NOT = { has_global_flag = zhx_system_initialised_v6 }
            }
            zhx_migrate_tianxia_system_v5_to_v6 = yes
        }
        if = {
            limit = {
                OR = {
                    NOT = { has_global_flag = zhx_gui_roster_initialised }
                    NOT = { has_global_flag = gdd_eoc_member_roster_initialised }
                }
            }
            zhx_build_gui_roster = yes
        }
    }

    option = {
        name = OK
    }
}

# The only delayed deadline event used by any public council. It carries no
# issue-specific result logic and refuses to touch a stale or closed ballot.
country_event = {
    id = zhx_system.23
    title = zhx_system.23.t
    desc = zhx_system.23.d
    picture = DIPLOMACY_eventPicture
    hidden = yes
    is_triggered_only = yes
    trigger = {
        zhx_is_tianzi = yes
        has_country_flag = zhx_council_phase_ballot_open
        has_country_flag = zhx_council_deadline_scheduled
        OR = {
            has_country_flag = zhx_council_kind_ritual_breakdown
            has_country_flag = zhx_council_kind_debate
        }
    }
    immediate = { zhx_resolve_tianxia_council = yes }
    option = { name = OK }
}

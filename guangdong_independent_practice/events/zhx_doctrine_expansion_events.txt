namespace = zhx_doctrine_expansion

# Triggered entry for the second doctrine triad. A dedicated country decision
# gives both players and AI an explicit, legible route into Dao/Bing/Zongheng
# instead of relying on a hidden random court invitation.
country_event = {
    id = zhx_doctrine_expansion.1
    title = zhx_doctrine_expansion.1.t
    desc = zhx_doctrine_expansion.1.d
    picture = DIPLOMACY_eventPicture
    is_triggered_only = yes

    trigger = { zhx_is_lijiao_country = yes }

    option = {
        name = zhx_doctrine_expansion.1.a
        country_event = { id = zhx_doctrine_expansion.10 }
        ai_chance = {
            factor = 10
            modifier = { factor = 2 is_at_war = no }
            modifier = { factor = 2 NOT = { war_exhaustion = 2 } }
            modifier = { factor = 2 army_size_percentage = 0.80 }
        }
    }
    option = {
        name = zhx_doctrine_expansion.1.b
        country_event = { id = zhx_doctrine_expansion.11 }
        ai_chance = {
            factor = 10
            modifier = { factor = 2 num_of_allies = 2 }
            modifier = { factor = 2 diplomatic_reputation = 1 }
        }
    }
    option = {
        name = zhx_doctrine_expansion.1.c
        country_event = { id = zhx_doctrine_expansion.12 }
        ai_chance = {
            factor = 10
            modifier = { factor = 2 army_size_percentage = 0.80 }
            modifier = { factor = 2 num_of_allies = 2 }
        }
    }
    option = {
        name = zhx_doctrine_expansion.1.e
        custom_tooltip = zhx_doctrine_expansion_postpone_tt
        hidden_effect = {
            add_country_modifier = {
                name = zhx_doctrine_change_cooldown
                duration = 730
            }
        }
        ai_chance = { factor = 0.25 }
    }
}

}

# The annual pulse sends this hidden event to every country, but only states
# already carrying a playable doctrine pass. The religion-change hook normally
# retires a departed state immediately; this branch remains a low-frequency
# safety net for ordinary gameplay.
country_event = {
    id = zhx_doctrine.90
    title = zhx_doctrine.90.t
    desc = zhx_doctrine.90.d
    picture = DIPLOMACY_eventPicture
    hidden = yes
    is_triggered_only = yes

    trigger = { zhx_has_doctrine = yes }

    immediate = {
        if = {
            limit = { zhx_is_lijiao_country = yes }
            zhx_yearly_doctrine_tick = yes
            zhx_ai_try_file_tianxia_debate_petition = yes
        }
        else = {
            zhx_retire_doctrine_system = yes
        }
    }

    option = { name = OK }
}

# Annual receipt calculator. The yearly scripted effect has already applied
# and clamped practice before reaching this hidden event. It records the real
# change, classifies both sides of the settlement, and opens player-only
# receipts when a 25/50/75 tier or the 70-point proposal boundary was crossed.
country_event = {
    id = zhx_doctrine.93
    title = zhx_doctrine.90.t
    desc = zhx_doctrine.90.d
    picture = DIPLOMACY_eventPicture
    hidden = yes
    is_triggered_only = yes

    trigger = { zhx_has_doctrine = yes }

    immediate = {
        set_variable = {
            which = zhx_doctrine_last_delta
            which = zhx_doctrine_practice
        }
        subtract_variable = {
            which = zhx_doctrine_last_delta
            which = zhx_doctrine_receipt_old_practice
        }

        set_variable = { which = zhx_doctrine_receipt_old_tier value = 0 }
        if = {
            limit = {
                check_variable = {
                    which = zhx_doctrine_receipt_old_practice
                    value = 75
                }
            }
            set_variable = { which = zhx_doctrine_receipt_old_tier value = 3 }
        }
        else_if = {
            limit = {
                check_variable = {
                    which = zhx_doctrine_receipt_old_practice
                    value = 50
                }
            }
            set_variable = { which = zhx_doctrine_receipt_old_tier value = 2 }
        }
        else_if = {
            limit = {
                check_variable = {
                    which = zhx_doctrine_receipt_old_practice
                    value = 25
                }
            }
            set_variable = { which = zhx_doctrine_receipt_old_tier value = 1 }
        }

        set_variable = { which = zhx_doctrine_receipt_new_tier value = 0 }
        if = {
            limit = {
                check_variable = { which = zhx_doctrine_practice value = 75 }
            }
            set_variable = { which = zhx_doctrine_receipt_new_tier value = 3 }
        }
        else_if = {
            limit = {
                check_variable = { which = zhx_doctrine_practice value = 50 }
            }
            set_variable = { which = zhx_doctrine_receipt_new_tier value = 2 }
        }
        else_if = {
            limit = {
                check_variable = { which = zhx_doctrine_practice value = 25 }
            }
            set_variable = { which = zhx_doctrine_receipt_new_tier value = 1 }
        }

        if = {
            limit = {
                ai = no
                # Tianxia proposal qualification has the higher information
                # priority. If this same settlement crosses 70, suppress the
                # ordinary tier receipt and let the dedicated .95 fire below.
                NOT = {
                    AND = {
                        zhx_is_tianxia_polity = yes
                        OR = {
                            AND = {
                                NOT = {
                                    check_variable = {
                                        which = zhx_doctrine_receipt_old_practice
                                        value = 70
                                    }
                                }
                                check_variable = {
                                    which = zhx_doctrine_practice
                                    value = 70
                                }
                            }
                            AND = {
                                check_variable = {
                                    which = zhx_doctrine_receipt_old_practice
                                    value = 70
                                }
                                NOT = {
                                    check_variable = {
                                        which = zhx_doctrine_practice
                                        value = 70
                                    }
                                }
                            }
                        }
                    }
                }
                OR = {
                    AND = {
                        NOT = {
                            check_variable = {
                                which = zhx_doctrine_receipt_old_tier
                                value = 1
                            }
                        }
                        check_variable = {
                            which = zhx_doctrine_receipt_new_tier
                            value = 1
                        }
                    }
                    AND = {
                        check_variable = {
                            which = zhx_doctrine_receipt_old_tier
                            value = 1
                        }
                        NOT = {
                            check_variable = {
                                which = zhx_doctrine_receipt_old_tier
                                value = 2
                            }
                        }
                        OR = {
                            NOT = {
                                check_variable = {
                                    which = zhx_doctrine_receipt_new_tier
                                    value = 1
                                }
                            }
                            check_variable = {
                                which = zhx_doctrine_receipt_new_tier
                                value = 2
                            }
                        }
                    }
                    AND = {
                        check_variable = {
                            which = zhx_doctrine_receipt_old_tier
                            value = 2
                        }
                        NOT = {
                            check_variable = {
                                which = zhx_doctrine_receipt_old_tier
                                value = 3
                            }
                        }
                        OR = {
                            NOT = {
                                check_variable = {
                                    which = zhx_doctrine_receipt_new_tier
                                    value = 2
                                }
                            }
                            check_variable = {
                                which = zhx_doctrine_receipt_new_tier
                                value = 3
                            }
                        }
                    }
                    AND = {
                        check_variable = {
                            which = zhx_doctrine_receipt_old_tier
                            value = 3
                        }
                        NOT = {
                            check_variable = {
                                which = zhx_doctrine_receipt_new_tier
                                value = 3
                            }
                        }
                    }
                }
            }
            set_country_flag = zhx_doctrine_tier_receipt_pending
            country_event = { id = zhx_doctrine.94 }
        }

        # Proposal qualification is a separate international threshold. Clear
        # both directions first so no stale marker can survive an unchanged
        # year, then dispatch at most one direction through an exclusive chain.
        clr_country_flag = zhx_doctrine_proposal_gained_pending
        clr_country_flag = zhx_doctrine_proposal_lost_pending
        if = {
            limit = {
                ai = no
                zhx_is_tianxia_polity = yes
                NOT = {
                    check_variable = {
                        which = zhx_doctrine_receipt_old_practice
                        value = 70
                    }
                }
                check_variable = { which = zhx_doctrine_practice value = 70 }
            }
            set_country_flag = zhx_doctrine_proposal_gained_pending
            country_event = { id = zhx_doctrine.95 }
        }
        else_if = {
            limit = {
                ai = no
                zhx_is_tianxia_polity = yes
                check_variable = {
                    which = zhx_doctrine_receipt_old_practice
                    value = 70
                }
                NOT = {
                    check_variable = { which = zhx_doctrine_practice value = 70 }
                }
            }
            set_country_flag = zhx_doctrine_proposal_lost_pending
            country_event = { id = zhx_doctrine.95 }
        }
    }

    option = { name = OK }
}

        ai = no
        zhx_is_tianxia_polity = yes
        zhx_has_doctrine = yes
        OR = {
            has_country_flag = zhx_doctrine_proposal_gained_pending
            has_country_flag = zhx_doctrine_proposal_lost_pending

            clr_country_flag = zhx_doctrine_proposal_gained_pending
            clr_country_flag = zhx_doctrine_proposal_lost_pending
        }
    }
}

# Conversion lifecycle cleanup. EU4 exposes no clear-school effect, so
# an obsolete eastern display mirror is replaced with a transparent sentinel.
# Keep the literal set here: nested scripted effects are not post-validated
# reliably by EU4 1.37.5 on macOS. Authoritative flags/variables are then reset.
country_event = {
    id = zhx_doctrine.92
    title = zhx_doctrine.90.t
    desc = zhx_doctrine.90.d
    picture = RELIGIOUS_CONVERSION_eventPicture
    hidden = yes
    is_triggered_only = yes

    trigger = {
        OR = {
            AND = {
                NOT = { zhx_is_lijiao_country = yes }
                zhx_has_any_doctrine_flag = yes
            }
            AND = {
                zhx_is_lijiao_country = yes
                NOT = { zhx_has_any_doctrine_flag = yes }
                has_religious_school = yes
            }
        }
    }

    immediate = {
        if = {
            limit = { religion_group = eastern }
            set_religious_school = {
                group = eastern
                school = zhx_no_doctrine_school
            }
        }
        zhx_clear_doctrine_system = yes
    }

    option = { name = OK }
}

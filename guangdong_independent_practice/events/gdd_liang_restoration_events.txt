# Random-without-replacement selector. A visited flag is removed before the
# validity check, so rejection or a 30-day invalid skip can never revisit it.
country_event = {
    id = gdd_liang_restoration.2
    title = none
    desc = none
    picture = DIPLOMACY_eventPicture
    hidden = yes
    is_triggered_only = yes

    trigger = {
        tag = CZH
        exists = yes
        has_global_flag = gdd_liang_restoration_intro_shown
        NOT = { has_global_flag = gdd_liang_restoration_succeeded }
        NOT = { has_global_flag = gdd_liang_restoration_failed }
    }

    immediate = {
        if = {
            limit = {
                any_country = { has_country_flag = gdd_liang_petition_roster }
            }
            random_country = {
                limit = { has_country_flag = gdd_liang_petition_roster }
                clr_country_flag = gdd_liang_petition_roster
                if = {
                    limit = { gdd_liang_is_valid_petition_recipient = yes }
                    country_event = { id = gdd_liang_restoration.10 }
                }
                else = {
                    CZH = {
                        country_event = { id = gdd_liang_restoration.2 days = 30 }
                    }
                }
            }
        }
        else = {
            set_global_flag = gdd_liang_restoration_failed
            country_event = { id = gdd_liang_restoration.30 }
        }
    }

    option = { name = OK }
}

# Monthly contract monitor; all other countries fail its event trigger cheaply.
country_event = {
    id = gdd_liang_restoration.90
    title = none
    desc = none
    picture = DIPLOMACY_eventPicture
    hidden = yes
    is_triggered_only = yes

    trigger = { has_country_flag = gdd_liang_restoration_patron_active }

    immediate = {
        if = {
            limit = { NOT = { exists = LGU } }
            gdd_liang_close_compact_neutrally = yes
        }
        else_if = {
            limit = { LGU = { NOT = { is_subject_of = ROOT } } }
            gdd_liang_close_compact_neutrally = yes
        }
        else_if = {
            limit = { LGU = { NOT = { is_march = yes } } }
            set_country_flag = gdd_liang_breach_was_revocation
            gdd_liang_apply_repudiation_with_liberty_desire = yes
            country_event = { id = gdd_liang_restoration.50 }
        }
        else_if = {
            limit = {
                gdd_liang_settlement_is_stable = yes
                NOT = { has_country_flag = gdd_liang_settlement_pending }
            }
            set_country_flag = gdd_liang_settlement_pending
            country_event = { id = gdd_liang_restoration.39 days = 30 }
        }
    }

    option = { name = OK }
}

namespace = zhx_debate

# Existing saves initialise the debate layer independently of the wider
# Tianxia migration chain. The event has one stable country anchor: CZH.
country_event = {
    id = zhx_debate.1
    title = zhx_debate.1.t
    desc = zhx_debate.1.d
    picture = DIPLOMACY_eventPicture
    hidden = yes
    is_triggered_only = yes

    trigger = {
        tag = CZH
        exists = yes
        has_global_flag = zhx_system_initialised_v4
        NOT = { has_global_flag = zhx_tianxia_debate_initialised_v1 }
    }

    immediate = { zhx_initialise_tianxia_debate = yes }
    option = { name = OK }
}

# Once at least two schools have a 70-practice exemplar, the Tianzi chooses
# which qualified pair will address the next international question.
country_event = {
    id = zhx_debate.2
    title = zhx_debate.2.t
    desc = zhx_debate.2.d
    picture = DIPLOMACY_eventPicture
    major = yes
    is_triggered_only = yes

    trigger = {
        zhx_can_convene_tianxia_debate = yes
    }

    option = {
        name = zhx_debate.2.a
        trigger = {
            zhx_tianxia_has_ru_proposal = yes
            zhx_tianxia_has_fa_proposal = yes
        }
        hidden_effect = { zhx_begin_tianxia_debate_ru_fa = yes }
        ai_chance = {
            factor = 10
            modifier = { factor = 3 has_country_flag = zhx_doctrine_ru }
            modifier = { factor = 3 has_country_flag = zhx_doctrine_fa }
        }
    }
    option = {
        name = zhx_debate.2.b
        trigger = {
            zhx_tianxia_has_ru_proposal = yes
            zhx_tianxia_has_mo_proposal = yes
        }
        hidden_effect = { zhx_begin_tianxia_debate_ru_mo = yes }
        ai_chance = {
            factor = 10
            modifier = { factor = 3 has_country_flag = zhx_doctrine_ru }
            modifier = { factor = 3 has_country_flag = zhx_doctrine_mo }
        }
    }
    option = {
        name = zhx_debate.2.c
        trigger = {
            zhx_tianxia_has_fa_proposal = yes
            zhx_tianxia_has_mo_proposal = yes
        }
        hidden_effect = { zhx_begin_tianxia_debate_fa_mo = yes }
        ai_chance = {
            factor = 10
            modifier = { factor = 3 has_country_flag = zhx_doctrine_fa }
            modifier = { factor = 3 has_country_flag = zhx_doctrine_mo }
        }
    }
}

# Annual scheduler: one Tianzi passes, and the event starts a debate only when
# the previous fifteen-year settlement has expired and two proposals qualify.
country_event = {
    id = zhx_debate.90
    title = zhx_debate.90.t
    desc = zhx_debate.90.d
    picture = DIPLOMACY_eventPicture
    hidden = yes
    is_triggered_only = yes

    trigger = {
        zhx_is_tianzi = yes
        has_global_flag = zhx_system_initialised_v4
    }

    immediate = {
        if = {
            limit = {
                NOT = { has_global_flag = zhx_tianxia_debate_initialised_v1 }
            }
            zhx_initialise_tianxia_debate = yes
        }
        if = {
            limit = { zhx_can_convene_tianxia_debate = yes }
            country_event = { id = zhx_debate.2 }
        }
    }
    option = { name = OK }
}

# Conservative AI entry.  The yearly public hook owns the five-year
# low-practice counter and rare 8% attempt; this event only dispatches
# the currently invited school and never uses an academy shortcut.
country_event = {
    id = zhx_doctrine_reform.50
    title = zhx_doctrine_reform.91.t
    desc = zhx_doctrine_reform.91.d
    picture = DIPLOMACY_eventPicture
    hidden = yes
    is_triggered_only = yes
    trigger = { zhx_doctrine_reform_ai_may_attempt = yes }
    immediate = {
        if = {
            limit = {
                zhx_doctrine_reform_has_ru_invited_root_with_five_years = yes
                NOT = { has_country_flag = zhx_doctrine_ru }
                zhx_doctrine_reform_can_target_ru = yes
            }
            zhx_doctrine_reform_begin_ru = yes
        }
        else_if = {
            limit = {
                zhx_doctrine_reform_has_fa_invited_root_with_five_years = yes
                NOT = { has_country_flag = zhx_doctrine_fa }
                zhx_doctrine_reform_can_target_fa = yes
            }
            zhx_doctrine_reform_begin_fa = yes
        }
        else_if = {
            limit = {
                zhx_doctrine_reform_has_mo_invited_root_with_five_years = yes
                NOT = { has_country_flag = zhx_doctrine_mo }
                zhx_doctrine_reform_can_target_mo = yes
            }
            zhx_doctrine_reform_begin_mo = yes
        }
        else_if = {
            limit = {
                zhx_doctrine_reform_has_dao_invited_root_with_five_years = yes
                NOT = { has_country_flag = zhx_doctrine_dao }
                zhx_doctrine_reform_can_target_dao = yes

            zhx_doctrine_reform_begin_dao = yes
        }
        else_if = {
            limit = {
                zhx_doctrine_reform_has_bing_invited_root_with_five_years = yes
                NOT = { has_country_flag = zhx_doctrine_bing }
                zhx_doctrine_reform_can_target_bing = yes

            zhx_doctrine_reform_begin_bing = yes
        }
        else_if = {
            limit = {
                zhx_doctrine_reform_has_zongheng_invited_root_with_five_years = yes
                NOT = { has_country_flag = zhx_doctrine_zongheng }
                zhx_doctrine_reform_can_target_zongheng = yes

            zhx_doctrine_reform_begin_zongheng = yes

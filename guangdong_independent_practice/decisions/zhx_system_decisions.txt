    # A player-facing authority operation. The cooldown is stored as a normal
    # country modifier so it survives saves and needs no date arithmetic.
    zhx_convene_grand_court = {
        major = yes

        potential = {
            has_global_flag = zhx_system_initialised_v3
            zhx_is_tianzi = yes
        }

        allow = {
            is_at_war = no
            stability = 0
            treasury = 100
            dip_power = 50
            NOT = { has_country_modifier = zhx_grand_court_cooldown }
        }

        effect = {
            add_treasury = -100
            add_dip_power = -50
            change_variable = {
                which = zhx_ritual_authority
                value = 10
            }
            zhx_clamp_ritual_authority = yes
            zhx_refresh_ritual_order = yes
            add_country_modifier = {
                name = zhx_grand_court_cooldown
                duration = 1825
            }
            country_event = { id = zhx_system.4 }
        }

        ai_will_do = {
            factor = 0
        }
    }
